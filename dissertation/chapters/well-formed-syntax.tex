This chapter constructs the well-typed syntax of $\ToS^\omega$ from the raw syntax of \cref{ch:raw-syntax}. \Cref{sec:natural-models,sec:shaped-syntax,sec:typings} are general: \cref{sec:natural-models} collects what we need of natural models; \cref{sec:shaped-syntax} identifies the category in which syntax over a set of shapes lives and describes, for a relative monad $T$ on a category of finite sets of shaped variables, the raw syntax as an object of it; and \cref{sec:typings} describes the judgement layers that can be imposed on the raw syntax, which we call typings. \Cref{sec:type-theories} defines the type theories over $T$, each specified by its raw types and a set of rules, and $\ToS^\omega$ as the type theory over the relative monad $\bE$ whose raw types are hypothetical judgements.

\section{Natural models}
\label{sec:natural-models}

For a presheaf $X : \cat{C}^{\op} \to \Set$, a morphism $\sigma : \Delta \to \Gamma \in \cat{C}$ and $x \in X(\Gamma)$, we write $x[\sigma] \coloneqq X(\sigma)(x) \in X(\Delta)$. The \emph{category of elements} $\cel{X}$ has as objects the pairs $(\Gamma, x)$ with $x \in X(\Gamma)$ and as morphisms $(\Delta, x[\sigma]) \to (\Gamma, x)$ the morphisms $\sigma : \Delta \to \Gamma$ of $\cat{C}$; the projection $\cel{X} \to \cat{C}$ is a discrete fibration. A natural transformation $f : Y \to X$ is the same as a morphism $\cel{f} : \cel{Y} \to \cel{X}$ over $\cat{C}$. We write $\yo : \cat{C} \to \Set^{\cat{C}^\op}$, $\yo\Gamma \coloneqq \cat{C}(\_, \Gamma)$, for the Yoneda embedding, and identify a natural transformation $\yo\Gamma \to X$ with an element of $X(\Gamma)$.

\begin{definition}[{\cite[Def.~1]{Awodey2018}}]\label{def:representable}
    A natural transformation $f : Y \to X$ of presheaves on $\cat{C}$ is \emph{representable} if for every object $\Gamma$ and every $A \in X(\Gamma)$ there are an object $\Gamma.A$, a morphism $p_A : \Gamma.A \to \Gamma$ and an element $\nu_A \in Y(\Gamma.A)$ such that 
    \[\begin{tikzcd}
    	{\yo(\Gamma.A)} & Y \\
    	{\yo(\Gamma)} & X
    	\arrow["{\nu_A}", from=1-1, to=1-2]
    	\arrow["{\yo(p_A)}"', from=1-1, to=2-1]
    	\arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=1-1, to=2-2]
    	\arrow["f", from=1-2, to=2-2]
    	\arrow["A"', from=2-1, to=2-2]
    \end{tikzcd}\]
    is a pullback square.
\end{definition}

\begin{lemma}\label{lem:right-adjoint}
    Let $f : Y \to X$ be a natural transformation of presheaves on $\cat{C}$.
    \begin{enumerate}
        \item For an object $\Gamma$, $A \in X(\Gamma)$, $p_A : \Gamma.A \to \Gamma$ and $\nu_A \in Y(\Gamma.A)$ with $f(\nu_A) = A[p_A]$, the square of \cref{def:representable} is a pullback iff for every $\sigma : \Delta \to \Gamma$ and $y \in Y(\Delta)$ with $f(y) = A[\sigma]$ there is a unique morphism $\langle \sigma, y \rangle : \Delta \to \Gamma.A$ with
        \[
            p_A \circ \langle \sigma, y \rangle = \sigma \qquad \text{and} \qquad \nu_A[\langle \sigma, y \rangle] = y.
        \]
        \item Let $R : \cel{X} \to \cel{Y}$ be a functor over $\cat{C}$ and $\varepsilon : \cel{f} \circ R \Rightarrow \id$ a natural transformation, and write $R(\Gamma, A) \eqqcolon (\Gamma.A, \nu_A)$ and $\varepsilon_{(\Gamma, A)} \eqqcolon p_A$, so that $f(\nu_A) = A[p_A]$. Then $\varepsilon$ is the counit of an adjunction $\cel{f} \dashv R$ iff every $(\Gamma.A, p_A, \nu_A)$ satisfies the condition of (1).
        \item $f$ is representable iff $\cel{f}$ has a right adjoint~\cite[Def.~3.8]{Uemura2023}.
    \end{enumerate}
\end{lemma}

\begin{proof}
    (1) Limits of presheaves are computed pointwise, and by the Yoneda lemma a natural transformation $\yo\Delta \to Z$ is an element of $Z(\Delta)$. So the square is a pullback iff for every $\Delta$ the map $\tau \mapsto (p_A \circ \tau, \nu_A[\tau])$ is a bijection from $\cat{C}(\Delta, \Gamma.A)$ onto $\{\, (\sigma, y) \mid \sigma : \Delta \to \Gamma,\ y \in Y(\Delta),\ f(y) = A[\sigma] \,\}$, which is the stated condition.

    (2) A morphism $(\Delta, y) \to (\Gamma.A, \nu_A)$ in $\cel{Y}$ is a morphism $\tau : \Delta \to \Gamma.A$ with $\nu_A[\tau] = y$, and a morphism $(\Delta, f(y)) \to (\Gamma, A)$ in $\cel{X}$ is a morphism $\sigma : \Delta \to \Gamma$ with $A[\sigma] = f(y)$. The natural transformation $\varepsilon$ is the counit of an adjunction iff for all $(\Delta, y)$ and $(\Gamma, A)$ the map $\tau \mapsto \varepsilon_{(\Gamma, A)} \circ \cel{f}(\tau) = p_A \circ \tau$ is a bijection between these sets, which is the condition of (1).

    (3) If $\cel{f}$ has a right adjoint, then by (2) and (1) each square is a pullback. Conversely, choose $(\Gamma.A, p_A, \nu_A)$ for every $(\Gamma, A)$; by (1), for a morphism $\sigma : (\Delta, A[\sigma]) \to (\Gamma, A)$ of $\cel{X}$ the morphism $\langle \sigma \circ p_{A[\sigma]}, \nu_{A[\sigma]} \rangle : (\Delta.A[\sigma], \nu_{A[\sigma]}) \to (\Gamma.A, \nu_A)$ is defined, and this makes $R$ a functor with the counit $\varepsilon_{(\Gamma, A)} = p_A$ natural; by (2) it is a right adjoint.
\end{proof}

\begin{lemma}[{\cite[Cor.~4]{Awodey2018}}]\label{lem:pullback}
    Let $f : Y \to X$ be a natural transformation with a right adjoint $(R, \varepsilon)$ of $\cel{f}$, and let $u : X' \to X$ be a natural transformation. Let $Y' \coloneqq X' \times_X Y$, so that $Y'(\Gamma) = \{\, (A', y) \mid A' \in X'(\Gamma),\ y \in Y(\Gamma),\ u(A') = f(y) \,\}$, and $f'(A', y) \coloneqq A'$. Then $\cel{f'}$ has the right adjoint
    \[
        R'(\Gamma, A') \coloneqq \big( \Gamma.u(A'),\ (A'[p_{u(A')}], \nu_{u(A')}) \big), \qquad \varepsilon'_{(\Gamma, A')} \coloneqq p_{u(A')},
    \]
    so that $\Gamma.A' = \Gamma.u(A')$, $p_{A'} = p_{u(A')}$, $\nu_{A'} = (A'[p_{A'}], \nu_{u(A')})$ and $\langle \sigma, (A'[\sigma], y) \rangle = \langle \sigma, y \rangle$. In particular, representable natural transformations are stable under pullback.
\end{lemma}

\begin{proof}
    We verify the condition of \cref{lem:right-adjoint}(1), which gives the claim by (2). $f'(\nu_{A'}) = A'[p_{A'}]$ by definition. Let $\sigma : \Delta \to \Gamma$ and $(B', y) \in Y'(\Delta)$ with $B' = A'[\sigma]$. Then $f(y) = u(B') = u(A')[\sigma]$, so $\langle \sigma, y \rangle : \Delta \to \Gamma.u(A')$ is defined, with $p_{A'} \circ \langle \sigma, y \rangle = \sigma$ and $\nu_{A'}[\langle \sigma, y \rangle] = (A'[p_{A'}][\langle \sigma, y \rangle], \nu_{u(A')}[\langle \sigma, y \rangle]) = (A'[\sigma], y)$. If $\tau : \Delta \to \Gamma.A'$ satisfies $p_{A'} \circ \tau = \sigma$ and $\nu_{A'}[\tau] = (A'[\sigma], y)$, then $\nu_{u(A')}[\tau] = y$, so $\tau = \langle \sigma, y \rangle$.
\end{proof}

\begin{definition}\label{def:natural-model}
    A \emph{natural model} $(\cat{C}, p)$ consists of a category $\cat{C}$ with a terminal object $\pt$, a natural transformation $p : \Tm \to \Ty$ of presheaves on $\cat{C}$, and a right adjoint $R$ of $P \coloneqq \cel{p}$ with counit $\varepsilon$. We write $\Gamma.A$, $p_A$, $\nu_A$ and $\langle \sigma, t \rangle$ as in \cref{lem:right-adjoint}, and $\Tm(\Gamma, A) \coloneqq \{\, t \in \Tm(\Gamma) \mid p(t) = A \,\}$; the elements of $\Ty(\Gamma)$ are the \emph{types} over $\Gamma$ and those of $\Tm(\Gamma, A)$ the \emph{terms} of $A$.
\end{definition}

By \cref{lem:right-adjoint}, a natural model is a representable $p$ together with a choice of the pullback squares of \cref{def:representable}, a \emph{structured} natural model in the terminology of~\cite[Def.~42]{AhrensLumsdaineNorth2024}.

\begin{lemma}\label{lem:pairing}
    In a natural model, for $\sigma : \Delta \to \Gamma.A$, $\tau : \Xi \to \Delta$ and $t \in \Tm(\Delta, A[\sigma'])$ with $\sigma' : \Delta \to \Gamma$:
    \begin{enumerate}
        \item $\sigma = \langle p_A \circ \sigma, \nu_A[\sigma] \rangle$;
        \item $\langle \sigma', t \rangle \circ \tau = \langle \sigma' \circ \tau, t[\tau] \rangle$;
        \item $\id_{\Gamma.A} = \langle p_A, \nu_A \rangle$.
    \end{enumerate}
\end{lemma}

\begin{proof}
    Each right-hand side is the unique morphism with the two defining properties, and the left-hand side has them: for (1) trivially; for (2) because $p_A \circ \langle \sigma', t \rangle \circ \tau = \sigma' \circ \tau$ and $\nu_A[\langle \sigma', t \rangle \circ \tau] = \nu_A[\langle \sigma', t \rangle][\tau] = t[\tau]$; (3) is the case $\sigma = \id$ of (1).
\end{proof}

\begin{definition}[{cf.~\cite[Thm.~2.3.2]{Newstead2018}}]\label{def:nm-morphism}
    A \emph{morphism} $F : (\cat{C}, p) \to (\cat{C}', p')$ of natural models consists of a functor $F : \cat{C} \to \cat{C}'$ with $F(\pt) = \pt$ and functors $F_\Ty : \cel{\Ty} \to \cel{\Ty'}$ and $F_\Tm : \cel{\Tm} \to \cel{\Tm'}$ over $F$, which commute with $P$, $P'$ and with the right adjoints,
    \[
    \begin{tikzcd}
        \cel{\Tm} \arrow[r, "F_\Tm"] \arrow[d, "P"'] & \cel{\Tm'} \arrow[d, "P'"] \\
        \cel{\Ty} \arrow[r, "F_\Ty"'] \arrow[d] & \cel{\Ty'} \arrow[d] \\
        \cat{C} \arrow[r, "F"'] & \cat{C}'
    \end{tikzcd}
    \qquad\qquad
    \begin{tikzcd}
        \cel{\Ty} \arrow[r, "F_\Ty"] \arrow[d, "R"'] & \cel{\Ty'} \arrow[d, "R'"] \\
        \cel{\Tm} \arrow[r, "F_\Tm"'] & \cel{\Tm'}
    \end{tikzcd}
    \]
    and such that $F_\Ty$ commutes with the counits, $F_\Ty \varepsilon = \varepsilon' F_\Ty$:
    \[
    \begin{tikzcd}[column sep=large]
        \cel{\Ty} \arrow[r, bend left=40, "PR", ""{name=T, below}] \arrow[r, bend right=40, "\id"', ""{name=B, above}] & \cel{\Ty} \arrow[r, "F_\Ty"] & \cel{\Ty'}
        \arrow[Rightarrow, from=T, to=B, "\varepsilon", shorten=1.5mm]
    \end{tikzcd}
    \;=\;
    \begin{tikzcd}[column sep=large]
        \cel{\Ty} \arrow[r, "F_\Ty"] & \cel{\Ty'} \arrow[r, bend left=40, "P'R'", ""{name=T, below}] \arrow[r, bend right=40, "\id"', ""{name=B, above}] & \cel{\Ty'}
        \arrow[Rightarrow, from=T, to=B, "\varepsilon'", shorten=1.5mm]
    \end{tikzcd}
    \]
    We write $F_\Ty(\Gamma, A) = (F\Gamma, FA)$ and $F_\Tm(\Gamma, t) = (F\Gamma, Ft)$. Natural models and their morphisms form a category $\mathsf{NM}$. We say that $F$ is injective, surjective or bijective \emph{on objects}, and faithful, full or fully faithful, in the usual sense; and that $F$ is \emph{fibrewise} injective, surjective or bijective \emph{on types}, resp.\ \emph{on terms}, if every $\Ty(\Gamma) \to \Ty'(F\Gamma)$, resp.\ every $\Tm(\Gamma, A) \to \Tm'(F\Gamma, FA)$, is so, and \emph{fibrewise} injective, surjective or bijective if it is so on both.
\end{definition}

\begin{lemma}\label{lem:nm-morphism}
    Let $(\cat{C}, p)$ and $(\cat{C}', p')$ be natural models, $F : \cat{C} \to \cat{C}'$ a functor with $F(\pt) = \pt$, and $F_\Ty : \cel{\Ty} \to \cel{\Ty'}$ and $F_\Tm : \cel{\Tm} \to \cel{\Tm'}$ functors over $F$, written as in \cref{def:nm-morphism}. Then $(F, F_\Ty, F_\Tm)$ is a morphism of natural models iff for all $\Gamma$, $A \in \Ty(\Gamma)$ and $t \in \Tm(\Gamma)$,
    \[
        p'(Ft) = F(pt), \qquad F(\Gamma.A) = F\Gamma.FA, \qquad F(\nu_A) = \nu_{FA}, \qquad F(p_A) = p_{FA}.
    \]
\end{lemma}

\begin{proof}
    On morphisms of the categories of elements, $F_\Ty$, $F_\Tm$, $P$ and $P'$ all act as $F$ or as the identity, so the first square of \cref{def:nm-morphism} commutes iff it does on objects, which is $p'(Ft) = F(pt)$. The second square commutes on objects iff $F(\Gamma.A) = F\Gamma.FA$ and $F(\nu_A) = \nu_{FA}$, since $F_\Tm R(\Gamma, A) = (F(\Gamma.A), F\nu_A)$ and $R' F_\Ty(\Gamma, A) = (F\Gamma.FA, \nu_{FA})$. Given these, the counit condition $F_\Ty \varepsilon = \varepsilon' F_\Ty$ is $F(p_A) = p_{FA}$, both sides being morphisms $F(\Gamma.A) = F\Gamma.FA \to F\Gamma$. It remains to see that, given the four equations, the second square commutes on morphisms. For a morphism $\sigma : (\Delta, A[\sigma]) \to (\Gamma, A)$ of $\cel{\Ty}$, we have $R\sigma = \langle \sigma \circ p_{A[\sigma]}, \nu_{A[\sigma]} \rangle$ by the proof of \cref{lem:right-adjoint}(3), and $F(R\sigma)$ is a morphism $F\Delta.FA[F\sigma] \to F\Gamma.FA$ with
    \begin{align*}
        p_{FA} \circ F(R\sigma) &= F(p_A \circ R\sigma) = F\sigma \circ p_{FA[F\sigma]},\\
        \nu_{FA}[F(R\sigma)] &= F(\nu_A[R\sigma]) = F(\nu_{A[\sigma]}) = \nu_{FA[F\sigma]},
    \end{align*}
    using $F(A[\sigma]) = FA[F\sigma]$, which is the naturality of $F_\Ty$; hence $F(R\sigma) = R'(F\sigma)$ by uniqueness.
\end{proof}

\begin{lemma}\label{lem:map-of-adjunctions}
    Let $F$ be a morphism of natural models. Then $F\langle \sigma, t \rangle = \langle F\sigma, Ft \rangle$ for all $\sigma : \Delta \to \Gamma$ and $t \in \Tm(\Delta, A[\sigma])$. Hence $F_\Tm$ commutes with the units, $F_\Tm \eta = \eta' F_\Tm$, and $(F_\Ty, F_\Tm)$ is a map of adjunctions from $P \dashv R$ to $P' \dashv R'$ in the sense of~\cite[\S IV.7]{MacLane1998}.
\end{lemma}

\begin{proof}
    By \cref{lem:nm-morphism}, $p_{FA} \circ F\langle \sigma, t \rangle = F(p_A \circ \langle \sigma, t \rangle) = F\sigma$ and $\nu_{FA}[F\langle \sigma, t \rangle] = F(\nu_A[\langle \sigma, t \rangle]) = Ft$, so $F\langle \sigma, t \rangle = \langle F\sigma, Ft \rangle$ by the uniqueness in \cref{lem:right-adjoint}(1). At $\sigma = \id_\Delta$ this says that $F_\Tm$ sends the unit $\langle \id_\Delta, t \rangle$ at $(\Delta, t)$ to the unit at $(F\Delta, Ft)$.
\end{proof}

By \cref{lem:nm-morphism}, morphisms of natural models are the homomorphisms of models of the essentially algebraic theory of natural models, whose sorts are objects, morphisms, types and terms; Newstead defines them so~\cite[Def.~2.3.1]{Newstead2018} and characterises them by the squares of \cref{def:nm-morphism} together with the commutation of $F_\Ty$ with the counits and of $F_\Tm$ with the units~\cite[Thm.~2.3.2]{Newstead2018}, the latter being redundant by \cref{lem:map-of-adjunctions}. Requiring instead that the mate $F_\Tm R \Rightarrow R' F_\Ty$ of the first square be an isomorphism, the Beck--Chevalley condition of~\cite[Def.~3.14]{Uemura2023}, rather than the identity, gives the weak morphisms of~\cite[Def.~2.3.9]{Newstead2018}, the pseudo maps of~\cite[Def.~42]{AhrensLumsdaineNorth2024}, and the morphisms of models of the basic dependent type theory of~\cite[Def.~4.14 and Thm.~4.12]{Uemura2023}.

\begin{lemma}\label{lem:iso}
    A morphism of natural models is an isomorphism in $\mathsf{NM}$ iff it is bijective on objects, fully faithful and fibrewise bijective.
\end{lemma}

\begin{proof}
    An inverse morphism inverts each of the four components. Conversely, let $F$ be bijective on objects, fully faithful and fibrewise bijective, so that each of its four components is a bijection, and let $G$ be given by the inverse maps. $G$ is a functor with $G(\pt) = \pt$, and $G_\Ty$, $G_\Tm$ are natural over $G$, because $F$ is a functor and $F_\Ty$, $F_\Tm$ are natural over $F$. The equations of \cref{lem:nm-morphism} hold for $G$ because they hold for $F$: for $\Gamma' = F\Gamma$ and $A' = FA$, $G(\Gamma'.A') = G(F(\Gamma.A)) = \Gamma.A = G\Gamma'.GA'$, and likewise $p(Gt) = G(p't)$, $G(\nu_{A'}) = \nu_{GA'}$ and $G(p_{A'}) = p_{GA'}$. So $G$ is a morphism, inverse to $F$.
\end{proof}

\begin{lemma}\label{lem:limits}
    The forgetful functor $U : \mathsf{NM} \to \Set^4$ recording the objects, morphisms, types and terms is faithful and creates limits.
\end{lemma}

\begin{proof}
    Let $K$ be a diagram in $\mathsf{NM}$ with vertices $K_i$ and $(L, \pi_i)$ a limit cone over $UK$ in $\Set^4$; up to the evident isomorphism, the elements of $L$ are the compatible families $(x_i)_i$ of objects, morphisms, types or terms of the $K_i$. If the $\pi_i$ are to be morphisms of natural models, the equations of \cref{lem:nm-morphism} force every piece of structure on $L$ to be computed componentwise: composition and $\pt$, $p$, the actions, $(\Gamma_i).(A_i) = (\Gamma_i.A_i)$, $p_{(A_i)} = (p_{A_i})$, $\nu_{(A_i)} = (\nu_{A_i})$ and $\langle (\sigma_i), (t_i) \rangle = (\langle \sigma_i, t_i \rangle)$. These families are compatible, because the morphisms of $K$ preserve the structure, and the axioms of a natural model, including the uniqueness of pairings in \cref{lem:right-adjoint}(1), hold because they hold componentwise; so there is exactly one cone over $K$ lying over $(L, \pi_i)$. It is a limit cone: a cone over $K$ with vertex $X$ induces a unique map $UX \to L$ in $\Set^4$, which is a morphism of natural models because its components into the $K_i$ are. Faithfulness is clear.
\end{proof}

\begin{lemma}\label{lem:mono}
    A morphism of natural models is a monomorphism of $\mathsf{NM}$ iff it is injective on objects, faithful and fibrewise injective.
\end{lemma}

\begin{proof}
    If $UF$ is injective in each component and $F \circ G = F \circ H$, then $UG = UH$, so $G = H$; thus $F$ is a monomorphism. Conversely, if $F$ is a monomorphism, its kernel pair is $(\id, \id)$; $U$ preserves this pullback by \cref{lem:limits}, so the kernel pair of $UF$ is $(\id, \id)$, which says that $UF$ is injective in each component; and injectivity on all types and terms is injectivity on objects together with fibrewise injectivity, since $F_\Ty(\Gamma, A) = (F\Gamma, FA)$ and $F_\Tm(\Gamma, t) = (F\Gamma, Ft)$.
\end{proof}

\begin{definition}\label{def:sub-natural-model}
    A \emph{sub-natural-model} of $(\cat{C}, p)$ is a subobject of $(\cat{C}, p)$ in $\mathsf{NM}$. By \cref{lem:mono,lem:nm-morphism}, it is given by a subcategory $\cat{W}$ of $\cat{C}$ containing $\pt$ together with subsets $\Ty_\cat{W}(\Gamma) \subseteq \Ty(\Gamma)$ and $\Tm_\cat{W}(\Gamma) \subseteq \Tm(\Gamma)$ for $\Gamma \in \cat{W}$, the images of the monomorphism, such that the restricted data form a natural model and the inclusion is a morphism: the subsets are closed under the actions of the morphisms of $\cat{W}$, $p(\Tm_\cat{W}(\Gamma)) \subseteq \Ty_\cat{W}(\Gamma)$, and for $\Gamma \in \cat{W}$ and $A \in \Ty_\cat{W}(\Gamma)$ we have $\Gamma.A \in \cat{W}$, $p_A \in \cat{W}$, $\nu_A \in \Tm_\cat{W}(\Gamma.A)$, and $\langle \sigma, t \rangle \in \cat{W}$ whenever $\sigma \in \cat{W}(\Delta, \Gamma)$ and $t \in \Tm_\cat{W}(\Delta) \cap \Tm(\Delta, A[\sigma])$.
\end{definition}

\begin{definition}[cf.~{\cite[Def.~2]{CastellanClairambaultDybjer2021}, \cite[Defs.~6 and~8]{AhrensLumsdaineNorth2024}}]\label{def:contextual}
    A natural model is \emph{contextual} if for every object $\Gamma$ there is a unique sequence $(A_1, \ldots, A_n)$, $n \geq 0$, with $A_i \in \Ty(\pt.A_1.\cdots.A_{i-1})$ and $\Gamma = \pt.A_1.\cdots.A_n$.

    The \emph{contextualization} $\Ctx(\cat{C}, p)$ of a natural model $(\cat{C}, p)$ is the natural model whose objects are the finite sequences $\vec{A} = (A_1, \ldots, A_n)$ with
    \[
        A_i \in \Ty(\langle A_1, \ldots, A_{i-1} \rangle), \qquad \text{where } \langle A_1, \ldots, A_k \rangle \coloneqq \pt.A_1.\cdots.A_k,
    \]
    and whose remaining data are inherited along $\langle \_ \rangle$:
    \[
        \Hom(\vec{A}, \vec{B}) \coloneqq \cat{C}(\langle \vec{A} \rangle, \langle \vec{B} \rangle), \qquad
        \Ty(\vec{A}) \coloneqq \Ty(\langle \vec{A} \rangle), \qquad
        \Tm(\vec{A}) \coloneqq \Tm(\langle \vec{A} \rangle),
    \]
    with the terminal object $()$, extension $\vec{A}.A \coloneqq (\vec{A}, A)$, and $p_A$, $\nu_A$, $\langle \sigma, t \rangle$ those of $\cat{C}$.
\end{definition}

\begin{proposition}\label{prop:contextualisation}
    $\Ctx(\cat{C}, p)$ is a contextual natural model, and $\vec{A} \mapsto \langle \vec{A} \rangle$ is a morphism of natural models that is fully faithful and fibrewise bijective. It is an isomorphism iff $(\cat{C}, p)$ is contextual, and it is~\cite[Prop.~48]{AhrensLumsdaineNorth2024} the counit of the adjunction
    \[\begin{tikzcd}
    	{\mathsf{CtxNM}} && {\mathsf{NM}.}
    	\arrow[""{name=0, anchor=center, inner sep=0}, "\iota", shift left=3, tail, from=1-1, to=1-3]
    	\arrow[""{name=1, anchor=center, inner sep=0}, "{\mathsf{Ctx}}", shift left=3, from=1-3, to=1-1]
    	\arrow["\dashv"{anchor=center, rotate=-90}, draw=none, from=0, to=1]
    \end{tikzcd}\]
\end{proposition}

\begin{proof}
    Identities, composition and the actions on types and terms are those of $\cat{C}$ at $\langle \vec{A} \rangle$, so $\Ctx(\cat{C}, p)$ is a category with presheaves $\Tm$, $\Ty$ and a natural transformation $p$; $()$ is terminal because $\langle () \rangle = \pt$ is. Since $\langle (\vec{A}, A) \rangle = \langle \vec{A} \rangle.A$, the right adjoint of $P$ restricts to one of $\cel{p}$ over $\Ctx(\cat{C}, p)$, with the same counit. Every object $(A_1, \ldots, A_n)$ is $().A_1.\cdots.A_n$, and $(\vec{A}, A) = (\vec{B}, B)$ iff $\vec{A} = \vec{B}$ and $A = B$, so the sequence is unique. The functor $\langle \_ \rangle$ preserves $\pt$, extension, $p_A$ and $\nu_A$ by definition, and is fully faithful and fibrewise bijective by construction. By \cref{lem:iso} it is an isomorphism iff it is bijective on objects as well, that is, iff every object of $\cat{C}$ is $\langle \vec{A} \rangle$ for exactly one sequence $\vec{A}$, which is contextuality.
\end{proof}

Contextuality is Cartmell's condition on contextual categories~\cite[Def.~44]{AhrensLumsdaineNorth2024}: contextual natural models are equivalent to contextual categories, later reformulated as Voevodsky's C-systems~\cite[\S4.2]{AhrensLumsdaineNorth2024}.

We write $\mathsf{CtxNM}$ for the full subcategory of $\mathsf{NM}$ on the contextual natural models. As the category of contextual categories, it is the category of models of an essentially algebraic theory whose sorts are the objects of each length and the morphisms between objects of given lengths~\cite[\S2.1]{KapulkinLumsdaine2018}, \cite[Rem.~2.6]{Voevodsky2014Subsystems}, so it has free objects: following \cite[Def.~3.9]{KapulkinLumsdaine2018}, we write $\free{\Gamma_n}$, $\free{\Gamma_n \vdash A}$ and $\free{\Gamma_n \vdash t : A}$ for the contextual natural models freely generated by an object of length $n$, by such an object with a type over it, and by these with a term of that type, so that a morphism $\free{\Gamma_n} \to (\cat{C}, p)$ is an object of length $n$ of $\cat{C}$, and likewise for the other two.

\begin{lemma}\label{lem:cxl-mono}
    A morphism of $\mathsf{CtxNM}$ is a monomorphism iff it is injective on objects, faithful and fibrewise injective. Hence, a subobject of a contextual natural model $(\cat{C}, p)$ in $\mathsf{CtxNM}$ is a sub-natural-model $\cat{W}$ of $(\cat{C}, p)$ closed under prefixes: $\Gamma \in \cat{W}$ and $A \in \Ty_\cat{W}(\Gamma)$ whenever $\Gamma.A \in \cat{W}$. We call these \emph{contextual sub-natural-models}.
\end{lemma}

\begin{proof}
    A sortwise injective morphism is a monomorphism because $U$ is faithful. Conversely, let $F : (\cat{C}, p) \to (\cat{C}', p')$ be a monomorphism of $\mathsf{CtxNM}$. If $F\Gamma = F\Gamma'$, then $\Gamma$ and $\Gamma'$ have the same length $n$, since $F$ preserves $\pt$ and extension and $\cat{C}'$ is contextual. The two morphisms $\free{\Gamma_n} \to (\cat{C}, p)$ choosing them have the same composite with $F$, $F \circ \Gamma = F \circ \Gamma'$, and hence, since $F$ is monic $\Gamma = \Gamma'$. The same argument works for types and for terms, with $\free{\Gamma_n \vdash A}$ and $\free{\Gamma_n \vdash t : A}$. Faithfulness follows by induction on the length of the codomain: morphisms into $\pt$ are unique, and if $F\sigma = F\sigma'$ for $\sigma, \sigma' : \Delta \to \Gamma.A$, then $F\sigma_0 = F\sigma'_0$ and $F\sigma_1 = F\sigma'_1$ by \cref{lem:nm-morphism}, so $\sigma_0 = \sigma'_0$ by induction and $\sigma_1 = \sigma'_1$ by injectivity on terms. Hence the image of a monomorphism out of a contextual natural model is a sub-natural-model as in \cref{def:sub-natural-model}, closed under the structure because the morphism preserves it, whose objects are the $F(\pt.A_1.\cdots.A_n) = \pt.FA_1.\cdots.FA_n$, hence closed under prefixes; conversely a sub-natural-model closed under prefixes is contextual, every object being uniquely $\pt.A_1.\cdots.A_n$ with $A_i \in \Ty_\cat{W}(\pt.A_1.\cdots.A_{i-1})$, and its inclusion is a monomorphism.
\end{proof}

\begin{proposition}\label{prop:trivial-fibrations}
    Let $I$ consist of the inclusions $\free{\Gamma_n} \to \free{\Gamma_n \vdash A}$ and $\free{\Gamma_n \vdash A} \to \free{\Gamma_n \vdash t : A}$ for $n \in \N$~\cite[Def.~3.10]{KapulkinLumsdaine2018}, and let $F : (\cat{C}, p) \to (\cat{C}', p')$ be a morphism of $\mathsf{CtxNM}$.
    \begin{enumerate}
        \item $F$ is fibrewise surjective iff it has the right lifting property against $I$. The fibrewise surjective morphisms, the \emph{trivial fibrations} of~\cite[Def.~3.11]{KapulkinLumsdaine2018}, form the right class of the weak factorisation system on $\mathsf{CtxNM}$ cofibrantly generated by $I$.
        \item If $F$ is fibrewise surjective, then $F$ is surjective on objects and full, and
        \begin{gather*}
            F(\Gamma.A) = F(\Gamma'.A') \iff F\Gamma = F\Gamma' \text{ and } F_\Ty(\Gamma, A) = F_\Ty(\Gamma', A'),\\
            F\langle \sigma, t \rangle = F\langle \sigma', t' \rangle \iff F\sigma = F\sigma' \text{ and } F_\Tm(t) = F_\Tm(t').
        \end{gather*}
        \item A fibrewise surjective morphism out of $(\cat{C}, p)$ is determined, up to a unique isomorphism, by its kernel on the total sets of types and of terms.
        \item A fibrewise surjective morphism is the coequaliser of its kernel pair, in $\mathsf{CtxNM}$ and in $\mathsf{NM}$; in particular it is a regular epimorphism of both.
    \end{enumerate}
\end{proposition}

\begin{proof}
    (1) A morphism $\free{\Gamma_n} \to (\cat{C}, p)$ is an object of $\cat{C}$, and a morphism $\free{\Gamma_n \vdash A} \to (\cat{C}', p')$ an object of $\cat{C}'$ with a type over it; so a lifting problem against the first map of $I$ is an object $\Gamma$ with a type $A' \in \Ty'(F\Gamma)$, and a solution is a type $A \in \Ty(\Gamma)$ with $FA = A'$. Likewise for terms. For the weak factorisation system we argue as \cite[Prop.~3.12]{KapulkinLumsdaine2018} does for contextual categories with additional structure. $\mathsf{CtxNM}$ is the category of models of a finitary essentially algebraic theory, hence locally finitely presentable~\cite[Thm.~3.36]{AdamekRosicky1994}, and the domains of the maps in $I$ are finitely presentable, being free on finitely many generators. Quillen's small object argument~\cite[Thm.~12.2.2]{Riehl2014} therefore factors every morphism as a transfinite composite of pushouts of maps in $I$ followed by a morphism with the right lifting property against $I$, and the retracts of the former together with the latter form a weak factorisation system.

    (2) Surjectivity on objects, by induction on the length: $\pt = F(\pt)$, and if $\Gamma' = F\Gamma$ and $A' \in \Ty'(\Gamma')$, then $A' = FA$ for some $A \in \Ty(\Gamma)$, so $\Gamma'.A' = F(\Gamma.A)$. Fullness, by induction on the length of $\Gamma$: for $\Gamma = \pt$ both hom-sets are singletons; a morphism $F\Delta \to F(\Gamma.A) = F\Gamma.FA$ is $\langle \sigma', t' \rangle$ with $\sigma' : F\Delta \to F\Gamma$ and $t' \in \Tm'(F\Delta, FA[\sigma'])$ by \cref{lem:pairing}; by induction $\sigma' = F\sigma$, so $FA[\sigma'] = F(A[\sigma])$, hence $t' = Ft$ for some $t \in \Tm(\Delta, A[\sigma])$, and $\langle \sigma', t' \rangle = F\langle \sigma, t \rangle$. The kernel: $F(\Gamma.A) = F\Gamma.FA$, and in a contextual natural model $\Gamma_1.A_1 = \Gamma_2.A_2$ implies $\Gamma_1 = \Gamma_2$ and $A_1 = A_2$, by uniqueness of the sequence; $F\langle \sigma, t \rangle = \langle F\sigma, Ft \rangle$ by \cref{lem:map-of-adjunctions}, and $\langle a, b \rangle = \langle a', b' \rangle$ implies $a = a'$ and $b = b'$, by applying $p_A \circ \_$ and $\nu_A[\_]$.

    (3) The claim is that if $F : (\cat{C}, p) \to (\cat{C}', p')$ and $G : (\cat{C}, p) \to (\cat{C}'', p'')$ are fibrewise surjective and
    \begin{gather*}
        F_\Ty(\Gamma, A) = F_\Ty(\Gamma', A') \iff G_\Ty(\Gamma, A) = G_\Ty(\Gamma', A'),\\
        F_\Tm(\Gamma, t) = F_\Tm(\Gamma', t') \iff G_\Tm(\Gamma, t) = G_\Tm(\Gamma', t'),
    \end{gather*}
    for all types and terms of $(\cat{C}, p)$, then $G = \theta \circ F$ for a unique isomorphism $\theta : (\cat{C}', p') \to (\cat{C}'', p'')$. By (2) and induction on the length, $F$ and $G$ have the same kernel on objects and on morphisms, so $\theta(Fx) \coloneqq Gx$ is well defined on objects, morphisms, types and terms; it is injective, and surjective because $G$ is surjective in every sort by (2). The equations of \cref{lem:nm-morphism} and the preservation of composition hold for $\theta$ because they hold for $F$ and $G$, and every object, morphism, type and term of $\cat{C}'$, composable pairs included, is in the image of $F$ by (2); the same applies to $\theta^{-1}$. Uniqueness holds because $\theta$ is determined on the image of $F$, which is everything by (2).

    (4) The kernel pair of $F$ in $\mathsf{CtxNM}$ is the contextualization $K$ of the pullback $(\cat{C}, p) \times_{(\cat{C}', p')} (\cat{C}, p)$ of $\mathsf{NM}$, which is computed sortwise by \cref{lem:limits}, since $\mathsf{CtxNM}$ is coreflective in $\mathsf{NM}$ by \cref{prop:contextualisation}. By (2) and induction on the length, two objects with $F\Gamma = F\Gamma'$ are $\pt.A_1.\cdots.A_n$ and $\pt.A'_1.\cdots.A'_n$ with $F_\Ty(\Gamma_{<i}, A_i) = F_\Ty(\Gamma'_{<i}, A'_i)$ for all $i$, so $(\Gamma, \Gamma')$ is the object $\langle (A_1, A'_1), \ldots, (A_n, A'_n) \rangle$ of $K$; hence every pair of objects, morphisms, types or terms with the same image under $F$ lies in $K$. If $G : (\cat{C}, p) \to (\cat{D}, r)$, for any natural model $(\cat{D}, r)$, equalises the two projections of $K$, it therefore identifies whatever $F$ identifies, and $Fx \mapsto Gx$ is a well-defined morphism $(\cat{C}', p') \to (\cat{D}, r)$, as in the proof of (3), the unique one through which $G$ factors. The kernel pair of $F$ in $\mathsf{NM}$ is computed sortwise by \cref{lem:limits} and consists of the same pairs, so the same argument shows that $F$ is its coequaliser in $\mathsf{NM}$.
\end{proof}

A weak factorisation system cofibrantly generated by a set $I$ of maps separates every morphism into a map built by attaching the cells of $I$, followed by a map in $I^{\pitchfork}$, along which every cell of $I$ lifts. The right class is therefore the notion of surjectivity, and hence of a quotient map, that the chosen cells determine, and the left class is the corresponding notion of free extension. The cells of $I$ in \cref{prop:trivial-fibrations} are the generators of the theory of natural models, so by (1) the trivial fibrations are the quotient maps relative to the generators.

In universal algebra, a quotient map is a homomorphism that is surjective on every sort and determined by its kernel. For contextual natural models, whose sorts depend on one another, \cref{prop:trivial-fibrations} says that the trivial fibrations are such maps: by (2) they are surjective in every sort, by (3) they are determined by their kernels, and by (4) they are regular. We are therefore justified to call them the \emph{quotient maps} of $\mathsf{CtxNM}$, cf.~\cite[Def.~7.4, Lemma~7.15]{Bocquet2023}.

\begin{definition}\label{def:subquotient}
    A \emph{subquotient} of a contextual natural model $\cat{X}$ is a contextual sub-natural-model $\cat{W} \hookrightarrow \cat{X}$ together with a trivial fibration $q : \cat{W} \twoheadrightarrow \cat{Q}$ with $\cat{Q}$ likewise contextual. In other words, it is a span of the form
    \[\begin{tikzcd}
    	{\cat{X}} & {\cat{W} } & {\cat{Q}}
    	\arrow[hook', from=1-2, to=1-1]
    	\arrow[two heads, from=1-2, to=1-3]
    \end{tikzcd}\]
    in $\mathsf{CtxNM}$.
    Two subquotients $(\cat{W}, q)$ and $(\cat{W}, q')$ with the same subobject are \emph{isomorphic} if $q' = \theta \circ q$ for an isomorphism $\theta : \cat{Q} \to \cat{Q}'$.
\end{definition}

\section{Shaped syntax}
\label{sec:shaped-syntax}

\begin{definition}\label{def:sorted-sets}
    Let $S$ be a set of \emph{shapes}. The category $\mathbb{F}{\downarrow}S$ has as objects the lists $\Xi = [s_1, \ldots, s_n]$ of shapes, and as morphisms $\rho : \Xi \to \Xi'$ the families of functions $\rho_s : (\Xi \ni s) \to (\Xi' \ni s)$, where $\Xi \ni s \coloneqq \{\, i \in \{1, \ldots, n\} \mid s_i = s \,\}$ is the set of \emph{variables} of $\Xi$ of shape $s$. The assignment $\Xi \mapsto (\Xi \ni s)_{s \in S}$ is a fully faithful functor $J_S : \mathbb{F}{\downarrow}S \to \Set^S$. For $S = 1$, $J_S$ is the inclusion $J_f : \mathbb{F} \to \Set$ of finite cardinals.
\end{definition}

\begin{definition}\label{def:shape-classifier}
    Let $S$ be a set of shapes. The \emph{natural model of shapes} is $\cat{B}_0(S) \coloneqq (\mathbf{1}, \id_S)$: the terminal category $\mathbf{1}$, with $\Ty = \Tm = S$ and $p$ the identity, so that every shape is a type with exactly one term; $\cel{p}$ is the identity functor, with the identity as right adjoint. The \emph{shape classifier} of $S$ is its contextualization $\cat{B}(S) \coloneqq \Ctx(\cat{B}_0(S))$: its objects are the lists over $S$, there is exactly one morphism between any two objects, $\Ty(\Xi) = S$, and every type has exactly one term.
\end{definition}

\begin{lemma}\label{lem:shaping}
    For a natural model $(\cat{C}, p)$, morphisms $(\cat{C}, p) \to \cat{B}_0(S)$ correspond bijectively to natural transformations $|\_| : \Ty \to S$ into the constant presheaf, which we call \emph{shapings}; a morphism $F$ between natural models with shapings lies over $\cat{B}_0(S)$ iff $|FA| = |A|$ for every type $A$, and is then called \emph{shape-preserving}. For contextual $(\cat{C}, p)$, morphisms to $\cat{B}_0(S)$ and to $\cat{B}(S)$ correspond bijectively.
\end{lemma}

\begin{proof}
    A morphism to $\cat{B}_0(S)$ consists of the unique functor to $\mathbf{1}$, a natural transformation $F_\Ty : \Ty \to S$, and $F_\Tm : \Tm \to S$, which by $p'(Ft) = F(pt)$ must be $F_\Ty \circ p$. The remaining equations of \cref{lem:nm-morphism} hold for every $F_\Ty$: $F(\Gamma.A) = \pt = F\Gamma.FA$ and $F(p_A) = \id = p_{FA}$ trivially, and $F(\nu_A) = F_\Ty(A[p_A]) = F_\Ty(A) = \nu_{FA}$ by naturality. A morphism $F$ lies over $\cat{B}_0(S)$ iff $|\_|' \circ F_\Ty = |\_|$. The last claim is the adjunction of \cref{prop:contextualisation}.
\end{proof}

\begin{definition}\label{def:shaped}
    For a set $S$ of shapes, a \emph{$S$-shaped} contextual natural model is a contextual natural model with a shaping $|\_| : \Ty \to S$, and $\mathsf{CtxNM}/\cat{B}(S)$ is the category of $S$-shaped contextual natural models and shape-preserving morphisms, the slice of $\mathsf{CtxNM}$ over $\cat{B}(S)$.
\end{definition}

\begin{remark}
    We will often omit the $S$- prefix and speak simply of shaped natural models.
\end{remark}

\begin{lemma}\label{lem:shaped-fiberwise-surjection}
    Let $\cat{X}$ be an object of $\mathsf{CtxNM}$. The weak factorisation system of \cref{prop:trivial-fibrations} induces one on the slice $\mathsf{CtxNM}/\cat{X}$: a morphism over $\cat{X}$ belongs to its left, resp.\ right, class iff its underlying morphism of $\mathsf{CtxNM}$ does. In particular, the trivial fibrations of $\mathsf{CtxNM}/\cat{X}$ are the morphisms over $\cat{X}$ that are fibrewise surjective. For $\cat{X} = \cat{B}(S)$, these are the fibrewise surjective shape-preserving morphisms, and for such a $q$, $q(\Gamma) = q(\Gamma')$ implies that $\Gamma$ and $\Gamma'$ have entrywise the same shapes.
\end{lemma}

\begin{proof}
    Let $\cat{L}'$ and $\cat{R}'$ be the classes of morphisms over $\cat{X}$ whose underlying morphisms lie in the left, resp.\ right, class of $\mathsf{CtxNM}$. Every morphism $f : Y \to Z$ over $\cat{X}$ factors as $f = r \circ l$ with $l \in \cat{L}'$ and $r \in \cat{R}'$: factor the underlying morphism in $\mathsf{CtxNM}$ and give the middle object the structure map obtained by composing that of $Z$ with $r$. A lift in $\mathsf{CtxNM}$ for a lifting problem over $\cat{X}$ between $l \in \cat{L}'$ and $r \in \cat{R}'$ is automatically over $\cat{X}$, since its composite with $r$ is; so $\cat{L}'$ and $\cat{R}'$ lift against each other, and as both are closed under retracts, the retract argument shows that each is the class of morphisms with the lifting property against the other. The same argument applies to any slice of any category with a weak factorisation system. For $\cat{X} = \cat{B}(S)$, morphisms over $\cat{X}$ are the shape-preserving ones by \cref{lem:shaping}, and $q(\Gamma) = q(\Gamma')$ gives $q_\Ty(\Gamma_{<i}, A_i) = q_\Ty(\Gamma'_{<i}, A'_i)$ for the entries, by \cref{prop:trivial-fibrations}(2) and induction on the length, hence $|A_i| = |qA_i| = |qA'_i| = |A'_i|$.
\end{proof}

\begin{definition}\label{def:shaped-subquotient}
    A subquotient of a $S$-shaped contextual natural model $X$ consists of a contextual sub-natural-model $\cat{W}$ of $X$ together with a fibrewise surjective shape-preserving morphism $q : \cat{W} \twoheadrightarrow \cat{Q}$ onto a $S$-shaped contextual natural model, i.e., it is given by a span in $\mathsf{CtxNM}/\mathsf{B}(S)$ of the form:
    \[\begin{tikzcd}
    	{\cat{X}} & {\cat{W} } & {\cat{Q}.}
    	\arrow[hook', from=1-2, to=1-1]
    	\arrow[two heads, from=1-2, to=1-3]
    \end{tikzcd}\]
\end{definition}

Intuitivelly, $\mathsf{CtxNM}/\cat{B}(S)$ is the category in which syntax over a set $S$ of shapes lives, since:
\begin{itemize}
    \item Syntax is contextual: a context is a list of types and a substitution into it a list of terms, and the pairing with which every natural model extends contexts is, in syntax, the only way in which contexts arise.
    \item Syntax is shaped: every type has a shape, the silent types which record equations will receive a shape $\top$ of their own, and no substitution changes a shape; so shapes are structure preserved by all morphisms.
\end{itemize}

We now restrict our attention to $\mathsf{CtxNM}/\cat{B}(S)$ that arise from a relative monad and a module over it. Recall that for a relative monad $T$ on $J$, a morphism $f : X \to Y$ of the Kleisli category $\Kl(T)$ is a morphism $f : JX \to TY$, with composition $g \odot f \coloneqq g^\sharp \circ f$. We first show that the opposite of the Kleisli category of a relative monad on $J_S$ is a shaped contextual natural model whose types are the shapes themselves, and then describe how modules of raw types and of raw equations over $T$ supply the types.

\begin{lemma}\label{lem:kleisli-coproducts}
    Let $T$ be a $J_S$-relative monad.
    \begin{enumerate}
        \item For a list $\Xi$ and a shape $s$, $\Kl(T)([s], \Xi) = \Set^S(J_S[s], T\Xi) \cong T(\Xi)_s$, naturally in $\Xi$, by evaluation at the unique variable of $[s]$.
        \item $\Kl(T)$ has finite coproducts: the empty list is initial, and for lists $\Xi$, $\Phi$ the list $\Xi \concat \Phi$, with the coprojections $\inl \coloneqq \eta \circ J_S(\inl)$ and $\inr \coloneqq \eta \circ J_S(\inr)$ induced by the inclusions of $\mathbb{F}{\downarrow}S$, is a coproduct: for every $\Delta$, the map
        \[
            \Kl(T)(\Xi \concat \Phi, \Delta) \to \Kl(T)(\Xi, \Delta) \times \Kl(T)(\Phi, \Delta), \qquad f \mapsto (f \odot \inl, f \odot \inr),
        \]
        is a bijection. We write $\langle f, g \rangle$ for its inverse, and $f \concat \Phi \coloneqq \langle \inl \odot f, \inr \rangle : \Xi \concat \Phi \to \Xi' \concat \Phi$ for $f : \Xi \to \Xi'$.
    \end{enumerate}
\end{lemma}

\begin{proof}
    (1) $J_S[s]$ has a single element, of shape $s$. (2) $J_S$ preserves the coproduct $\Xi \concat \Phi$ of $\mathbb{F}{\downarrow}S$, so $f \mapsto (f \circ J_S(\inl), f \circ J_S(\inr))$ is a bijection $\Set^S(J_S(\Xi \concat \Phi), T\Delta) \to \Set^S(J_S\Xi, T\Delta) \times \Set^S(J_S\Phi, T\Delta)$, and by the unit law $f \odot \inl = f^\sharp \circ \eta \circ J_S(\inl) = f \circ J_S(\inl)$, likewise for $\inr$. The empty list has no variables, so $\Kl(T)(\emptyar, \Delta)$ is a singleton.
\end{proof}

\begin{proposition}\label{prop:nm-of-monad}
    Let $T$ be a relative monad on $J_S$, and let $\cat{C}_T \coloneqq \Kl(T)^\op$. Define presheaves on $\cat{C}_T$ by
    \[
        \Ty_T(\Xi) \coloneqq S, \qquad \Tm_T(\Xi) \coloneqq \textstyle\coprod_{s \in S} T(\Xi)_s, \qquad t[\sigma] \coloneqq \sigma^\sharp(t),
    \]
    with the trivial action on $\Ty_T$, and let $p_T : \Tm_T \to \Ty_T$ send $T(\Xi)_s$ to $s$. For a list $\Xi$ and a shape $s$, put
    \begin{gather*}
        \Xi.s \coloneqq \Xi \concat [s], \qquad p_s \coloneqq \inl : \Xi \to \Xi \concat [s],\\
        \nu_s \coloneqq \inr \in \Kl(T)([s], \Xi \concat [s]) = T(\Xi \concat [s])_s.
    \end{gather*}
    Then
    \[\begin{tikzcd}
        {\yo(\Xi \concat [s])} & \Tm_T \\
        {\yo(\Xi)} & \Ty_T
        \arrow["{\nu_s}", from=1-1, to=1-2]
        \arrow["{\yo(p_s)}"', from=1-1, to=2-1]
        \arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=1-1, to=2-2]
        \arrow["{p_T}", from=1-2, to=2-2]
        \arrow["s"', from=2-1, to=2-2]
    \end{tikzcd}\]
    is a pullback square, so $p_T$ is representable, and with these squares $(\cat{C}_T, p_T)$ is a $S$-shaped contextual natural model with terminal object the empty list, whose pairing is $\langle \sigma, t \rangle = \langle f, t \rangle : \Xi \concat [s] \to \Delta$ for $\sigma : \Delta \to \Xi$ given by $f$ and $t \in T(\Delta)_s = \Kl(T)([s], \Delta)$.
\end{proposition}

\begin{proof}
    $\Tm_T$ is a presheaf by the laws of $\Kl(T)$, and $p_T$ is natural because $f^\sharp$ preserves shapes. The empty list is initial in $\Kl(T)$ by \cref{lem:kleisli-coproducts}, hence terminal in $\cat{C}_T$. Under the identification $T(\Gamma)_s = \Kl(T)([s], \Gamma)$ of \cref{lem:kleisli-coproducts}(1), the action $t \mapsto t[\sigma] = g^\sharp(t)$ of $\sigma : \Delta \to \Gamma$ given by $g : \Gamma \to \Delta$ is $h \mapsto g \odot h$, since $(g \odot h)(x) = g^\sharp(h(x))$ for the variable $x$ of $[s]$; in particular $\nu_s[\sigma] = g \odot \inr$ for $\sigma : \Delta \to \Xi \concat [s]$ given by $g$. The square commutes, since $p_T(\nu_s) = s = s[p_s]$, $\Ty_T$ being constant. Pullbacks of presheaves are computed pointwise, so fix a list $\Delta$. By the Yoneda lemma, at $\Delta$ the square is
    \[\begin{tikzcd}[column sep=large]
        {\Kl(T)(\Xi \concat [s], \Delta)} & {\coprod_{r \in S} T(\Delta)_r} \\
        {\Kl(T)(\Xi, \Delta)} & S
        \arrow["{g \mapsto g \odot \inr}", from=1-1, to=1-2]
        \arrow["{g \mapsto g \odot \inl}"', from=1-1, to=2-1]
        \arrow["{p_T}", from=1-2, to=2-2]
        \arrow["{f \mapsto s}"', from=2-1, to=2-2]
    \end{tikzcd}\]
    and the pullback of its lower right corner is $\{\, (f, t) \mid f \in \Kl(T)(\Xi, \Delta),\ p_T(t) = s \,\} = \Kl(T)(\Xi, \Delta) \times \Kl(T)([s], \Delta)$. So the square is a pullback iff $g \mapsto (g \odot \inl, g \odot \inr)$ is a bijection $\Kl(T)(\Xi \concat [s], \Delta) \to \Kl(T)(\Xi, \Delta) \times \Kl(T)([s], \Delta)$, which is \cref{lem:kleisli-coproducts}(2). By \cref{lem:right-adjoint}, the chosen squares define a right adjoint of $\cel{p_T}$ with counit $p_s$, and $\langle \sigma, t \rangle$ is the unique $g$ with $g \odot \inl = f$ and $g \odot \inr = t$, which is $\langle f, t \rangle$. Every list is uniquely $\emptyar.s_1.\cdots.s_n$, so the model is contextual.
\end{proof}

\begin{proposition}\label{prop:rmnd-shaped}
    The assignment $T \mapsto \cat{C}_T$ defines a fully faithful functor $$\cat{C}_{(\_)} : \mathsf{RMnd}(J_S) \to \mathsf{CtxNM}/\cat{B}(S)$$ whose essential image consists of $S$-shaped contextual natural models whose shaping is trivial in the sense that $\Ty(\Gamma) \to S$ is a bijection for every $\Gamma$. 
\end{proposition}

\begin{proof}
    Write $x_i$ for the $i$-th variable of $\Xi = [s_1, \ldots, s_n]$, $\pi_i$ for the composite of the projections $p_{s_j}$, $j > i$, and $\sigma_f : \Delta \to \Xi$ for the morphism of $\cat{C}_T$ given by $f : J_S\Xi \to T\Delta$. As $\nu_s = \inr$ and $p_s = \inl$, the unit law gives $\eta(x_i) = \nu_{s_i}[\pi_i]$ and $\eta(x_i)[\sigma_f] = f(x_i)$: the variables are the weakened generic terms, and a morphism into $\Xi$ is determined by its action on them.

    \emph{Functor.} A morphism $\varphi : T \to T'$ of relative monads (\cref{sec:lambda}) induces $\cat{C}_\varphi$, the identity on objects and types, sending $\sigma_f$ to $\sigma_{\varphi \circ f}$ and acting by $\varphi$ on terms. As $\varphi$ preserves units and Kleisli extensions, this is a functor, compatible with the actions, $\varphi(t[\sigma_f]) = (\varphi \circ f)^\sharp(\varphi t) = \varphi(t)[\sigma_{\varphi \circ f}]$, and preserving $p_s = \sigma_{\eta \circ J_S(\inl)}$ and $\nu_s$, the unit at the last variable; so it is a shape-preserving morphism by \cref{lem:nm-morphism}.

    \emph{Full faithfulness.} $\varphi$ is the action of $\cat{C}_\varphi$ on terms. Conversely, a shape-preserving $F : \cat{C}_T \to \cat{C}_{T'}$ is the identity on types, both shapings being $\id_S$, hence on objects by \cref{lem:nm-morphism}, and on terms a family $\varphi_\Xi : T\Xi \to T'\Xi$. As $F$ preserves $\nu$, $p$ and the actions, $\varphi(\eta(x_i)) = \eta'(x_i)$, and $F\sigma_f$ acts on the variables by $\varphi \circ f$, so $F\sigma_f = \sigma_{\varphi \circ f}$ and $\varphi(f^\sharp t) = \varphi(t)[F\sigma_f] = (\varphi \circ f)^\sharp(\varphi t)$. Thus $\varphi$ is a morphism and $F = \cat{C}_\varphi$.

    \emph{Essential image.} The condition holds for $\cat{C}_T$ and is invariant under isomorphism over $\cat{B}(S)$ by \cref{lem:iso}. Conversely, let $X$ have trivial shaping, and identify its types with $S$, so that $s[\sigma] = s$. By contextuality, its objects are the $\langle \Xi \rangle \coloneqq \pt.s_1.\cdots.s_n$, each exactly once, and by \cref{lem:pairing} and induction on $n$, $\sigma \mapsto (v_i[\sigma])_i$ with $v_i \coloneqq \nu_{s_i}[\pi_i]$ is a bijection $X(\langle \Delta \rangle, \langle \Xi \rangle) \to \prod_i \Tm_X(\langle \Delta \rangle, s_i)$. Hence $T_X(\Xi)_s \coloneqq \Tm_X(\langle \Xi \rangle, s)$, with $\eta(x_i) \coloneqq v_i$ and $f^\sharp(t) \coloneqq t[\sigma_f]$ for the morphism $\sigma_f$ with $v_i[\sigma_f] = f(x_i)$, is a relative monad on $J_S$: the equations $\sigma_\eta = \id$, $v_i[\sigma_f] = f(x_i)$ and $\sigma_{g^\sharp \circ f} = \sigma_f \circ \sigma_g$ give its three laws. The assignments $\Xi \mapsto \langle \Xi \rangle$ and $\sigma_f \mapsto \sigma_f$, with the identities on types and terms, form a shape-preserving morphism $\cat{C}_{T_X} \to X$ preserving $p_s$ and $\nu_s$, which is bijective on objects, fully faithful and fibrewise bijective, hence an isomorphism by \cref{lem:iso}.
\end{proof}

\begin{definition}\label{def:silent-types}
    Let $(\cat{C}, p)$ be a natural model and $E$ a presheaf on $\cat{C}$. The natural model $(\cat{C}, p)$ \emph{with silent types $E$} is
    \[
        p \sqcup \id_E : \Tm \sqcup E \longrightarrow \Ty \sqcup E,
    \]
    the coproduct of $p$ with the identity of $E$, coproducts of presheaves being computed pointwise. The elements of $E$ are the \emph{silent} types; each $e \in E(\Gamma)$ is a type over $\Gamma$ with exactly one term, the element $e$ of the summand $E$ of $\Tm \sqcup E$, which we write $\star$.
\end{definition}

\begin{lemma}\label{lem:silent-types}
    The map $p \sqcup \id_E$ is representable, with right adjoint $R \sqcup \id$ under the identifications $\cel{(\Tm \sqcup E)} = \cel{\Tm} \sqcup \cel{E}$ and $\cel{(\Ty \sqcup E)} = \cel{\Ty} \sqcup \cel{E}$, so that $(\cat{C}, p \sqcup \id_E)$ is a natural model. Explicitly, at a type of $(\cat{C}, p)$ the data are those of $(\cat{C}, p)$, and at a silent type $e \in E(\Gamma)$
    \[
        \Gamma.e = \Gamma, \qquad p_e = \id_\Gamma, \qquad \nu_e = \star, \qquad \langle \sigma, \star \rangle = \sigma.
    \]
    In particular $(\cat{C}, p \sqcup \id_E)$ is not contextual unless $E$ is empty.
\end{lemma}

\begin{proof}
    The category of elements of a coproduct of presheaves is the coproduct of their categories of elements, since a morphism of $\cel{(X \sqcup Y)}$ lies in the summand of its codomain. Hence $\cel{(p \sqcup \id_E)} = \cel{p} \sqcup \id_{\cel{E}}$, and a coproduct of functors with right adjoints $R$ and $\id$ has the right adjoint $R \sqcup \id$, with the coproduct of the counits. The explicit form at $e$ unwinds the identity adjunction: $\sigma$ is the unique morphism $\tau : \Delta \to \Gamma$ with $\id_\Gamma \circ \tau = \sigma$ and $\star[\tau] = \star$, as in \cref{lem:right-adjoint}(1). Since $\Gamma.e = \Gamma$, the sequence presenting $\Gamma$ is not unique when $E(\Gamma) \neq \emptyset$.
\end{proof}

\begin{definition}\label{def:module}
    Let $T$ be a relative monad on $J : \cat{J} \to \cat{C}$. A \emph{$T$-module} in a category $\cat{D}$ is a functor $M : \Kl(T) \to \cat{D}$; explicitly, an assignment of an object $MX$ to each object $X$ of $\cat{J}$ and of a morphism $f^M : MX \to MY$ to each $f : JX \to TY$, such that $\eta_X^M = \id_{MX}$ and $(g \odot f)^M = g^M \circ f^M$. A \emph{morphism} of $T$-modules is a natural transformation. In particular, $T$ itself is a $T$-module in $\cat{C}$, with $f^T \coloneqq f^\sharp$, and a $T$-module in $\Set$ is the same as a presheaf on $\cat{C}_T$.
\end{definition}

\begin{remark}
    To see how this definition corresponds to the usual notion of modules, let $T : \cat{C} \to \cat{D}$ be a relative monad over $J : \cat{C} \to \cat{D}$ and let $M : \Kl(T) \to \Set$ be a module over $T$. The elements of $M$ are indexed over the objects of $\cat{C}$. In this settings, the scalars are the substitutions $\sigma : JX \to TY$, so that scalar multiplication is construed as the family of operations
    $$\_\cdot\_ : \{X,Y : \cat{C}\} \to \Hom_{\cat{D}}(JX,TY) \times MX \to MY.$$
    The module laws are then precisely the functoriality of $M$: we have $\eta_X\cdot x = x$ and $g \cdot (f \cdot x) = (g \odot f) \cdot x$. 
\end{remark}

\begin{definition}\label{def:raw-types}
    Let $T$ be a relative monad on $J_S$. A \emph{module of raw types} over $T$ is a $T$-module $M$ in $\Set^S$. We will regard $M$ as the presheaf $\Xi \mapsto \coprod_{s \in S} M(\Xi)_s$ on $\cat{C}_T$, with action $A[\sigma] \coloneqq f^M(A)$ for $\sigma$ given by $f$, together with the natural transformation $\fst : M \to \Ty_T$, the first projection, sending $M(\Xi)_s$ to $s$, the \emph{shape}. A \emph{module of raw equations} over $T$ is a $T$-module $E$ in $\Set$, that is, a presheaf on $\cat{C}_T$; we write $e[\sigma] \coloneqq f^E(e)$.
\end{definition}

\begin{definition}\label{def:raw-natural-model}
    Let $T$ be a relative monad on $J_S$, $M$ a module of raw types and $E$ a module of raw equations over $T$. Let $p_M : M \times_{\Ty_T} \Tm_T \to M$ be defined by
    \[\begin{tikzcd}
    	{M\times_{\Ty_T}\Tm_T} & {\Tm_T} \\
    	M & {\Ty_T},
    	\arrow[from=1-1, to=1-2]
    	\arrow["{p_M}"', from=1-1, to=2-1]
    	\arrow["\lrcorner"{anchor=center, pos=0.125}, draw=none, from=1-1, to=2-2]
    	\arrow["{p_T}", from=1-2, to=2-2]
    	\arrow["{\fst}"', from=2-1, to=2-2]
    \end{tikzcd}\]
    i.e.\ as the base change of $p_T$ along $|\_|$, with the right adjoint given by \cref{lem:pullback}, so that $(\cat{C}_T, p_M)$ is a natural model. The \emph{raw natural model} of $T$ with types $M$ and equations $E$ is
    \[
        \Raw(T, M, E) \coloneqq \Ctx(\cat{C}_T, p_M \sqcup \id_E),
    \]
    the contextualization of $(\cat{C}_T, p_M)$ with silent types $E$, a contextual natural model by \cref{lem:silent-types,prop:contextualisation}. It is shaped by
    $$|\_| := \left<\fst, \_ \mapsto \top\right> : M \sqcup E \to S^+,$$
    so it is an object of $\mathsf{CtxNM}/\cat{B}(S^+)$. Its objects are the \emph{raw contexts}.
\end{definition}

\begin{remark}
    Read type-theoretically, $M$ says what variables can be declared, $E$ says what can be assumed without declaring a variable, and $T$ says what can be substituted for the variables, while $S$ determines the possible binging arities of the variables.
\end{remark}

\begin{proposition}\label{prop:raw-natural-model}
    Define the \emph{underlying list} $(\_)^{\downarrow} : \Ob \Raw(T, M, E) \to \Ob \Kl(T)$ by recursion on the length of $\Gamma = (A_1, \ldots, A_n)$: $()^{\downarrow} \coloneqq \emptyar$, $(\Gamma, A)^{\downarrow} \coloneqq \Gamma^{\downarrow} \concat [|A|]$ for $A \in M$, and $(\Gamma, e)^{\downarrow} \coloneqq \Gamma^{\downarrow}$ for $e \in E$. Then $\Raw(T, M, E)$ is as follows.
    \begin{enumerate}
        \item Objects are the sequences $\Gamma = (A_1, \ldots, A_n)$ with $A_i \in M(\Gamma_{<i}^{\downarrow}) \sqcup E(\Gamma_{<i}^{\downarrow})$, where $\Gamma_{<i} \coloneqq (A_1, \ldots, A_{i-1})$. A morphism $\sigma : \Delta \to \Gamma$ is a morphism $f : \Gamma^{\downarrow} \to \Delta^{\downarrow}$ of $\Kl(T)$; identities and composition are those of $\Kl(T)^\op$.
        \item $\Ty(\Gamma) = M(\Gamma^{\downarrow}) \sqcup E(\Gamma^{\downarrow})$, with $A[\sigma] = f^M(A)$ and $e[\sigma] = f^E(e)$.
        \item $\Tm(\Gamma, A) = T(\Gamma^{\downarrow})_{|A|}$ for $A \in M(\Gamma^{\downarrow})$, with $t[\sigma] = f^\sharp(t)$, and $\Tm(\Gamma, e) = \{\star\}$ for $e \in E(\Gamma^{\downarrow})$.
        \item $\Gamma.A = (\Gamma, A)$. For $A \in M(\Gamma^{\downarrow})$ with $|A| = s$: $p_A$ is $\inl : \Gamma^{\downarrow} \to \Gamma^{\downarrow} \concat [s]$, $\nu_A = \inr \in T(\Gamma^{\downarrow} \concat [s])_s$, and $\langle \sigma, t \rangle$ is $\langle f, t \rangle : \Gamma^{\downarrow} \concat [s] \to \Delta^{\downarrow}$. For $e \in E(\Gamma^{\downarrow})$: $p_e$ is $\id_{\Gamma^{\downarrow}}$, $\nu_e = \star$, and $\langle \sigma, \star \rangle = \sigma$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    By induction on $n$, $\langle (A_1, \ldots, A_n) \rangle = (A_1, \ldots, A_n)^{\downarrow}$ in $\cat{C}_T$, since $\Xi.A = \Xi.|A| = \Xi \concat [|A|]$ for $A \in M(\Xi)$ by \cref{lem:pullback,prop:nm-of-monad}, and $\Xi.e = \Xi$ for $e \in E(\Xi)$ by \cref{lem:silent-types}. The rest unfolds \cref{def:contextual,lem:pullback,lem:silent-types,prop:nm-of-monad}: the fibre of $p_M$ over $A \in M(\Xi)$ is $\{(A, t) \mid p_T(t) = |A|\} \cong \Tm_T(\Xi, |A|) = T(\Xi)_{|A|}$, and we identify $(A, t)$ with $t$; the fibre over $e \in E(\Xi)$ is $\{\star\}$.
\end{proof}

\begin{remark}
    Read as a C-system, for $E = \emptyset$ and $S = 1$, $\Raw(T, M, \emptyset)$ is Voevodsky's C-system $\Csys(T, M)$ of a module over a relative monad~\cite[\S6]{Voevodsky2016Relative}.
\end{remark}

\begin{remark}\label{cor:sort-degeneracy}
    By \cref{prop:raw-natural-model}, the types over $\Gamma$ depend only on $\Gamma^{\downarrow}$, and the terms of $A$ only on $\Gamma^{\downarrow}$ and $|A|$. Type-theoretically, this says precisely that the raw syntax is scoped, but not typed. Moreover, $\Hom(\Delta, \Gamma) = \Kl(T)(\Gamma^{\downarrow}, \Delta^{\downarrow})$ depends only on the underlying lists, so that any two raw contexts with the same underlying list are isomorphic, via the identity of that list; we call these the \emph{canonical isomorphisms} of $\Raw(T, M, E)$.
\end{remark}

\section{Typings}
\label{sec:typings}

In this section, $T$ is a relative monad on $J_S$, $M$ a module of raw types and $E$ a module of raw equations over $T$, and $\Raw \coloneqq \Raw(T, M, E)$. The letters $\Gamma, \Delta, \Xi$ range over raw contexts, $\sigma, \tau$ over morphisms of $\Raw$, $A, A'$ over types and $t, t'$ over terms. For $\sigma : \Delta \to \Gamma.A$ we write $\sigma_0 \coloneqq p_A \circ \sigma : \Delta \to \Gamma$ and $\sigma_1 \coloneqq \nu_A[\sigma] \in \Tm(\Delta, A[\sigma_0])$, so that $\sigma = \langle \sigma_0, \sigma_1 \rangle$ by \cref{lem:pairing}.

\begin{definition}\label{def:raw-judgements}
    The \emph{raw judgements} of $\Raw$ are the elements of the sets
    \begin{align*}
        B &\coloneqq \{\, (\Gamma, A) \mid A \in \Ty(\Gamma) \,\},\\
        \widetilde{B} &\coloneqq \{\, (\Gamma, A, t) \mid t \in \Tm(\Gamma, A) \,\},\\
        B_{\mathrm{eq}} &\coloneqq \{\, (\Gamma, A, A') \mid A, A' \in \Ty(\Gamma),\ |A| = |A'| \,\},\\
        \widetilde{B}_{\mathrm{eq}} &\coloneqq \{\, (\Gamma, A, t, t') \mid t, t' \in \Tm(\Gamma, A) \,\},
    \end{align*}
    written $\Gamma \vdash A$, $\Gamma \vdash t : A$, $\Gamma \vdash A \equiv A'$ and $\Gamma \vdash t \equiv t' : A$, respectively. A \emph{set of judgements} is a subset $\jdg \subseteq B \sqcup \widetilde{B} \sqcup B_{\mathrm{eq}} \sqcup \widetilde{B}_{\mathrm{eq}}$. When $\jdg$ is fixed, a raw judgement written on its own asserts membership in $\jdg$.
\end{definition}

\begin{definition}\label{def:derived-forms}
    Let $\jdg$ be a set of judgements. For raw contexts $\Gamma, \Gamma', \Delta$ and morphisms $\sigma, \sigma' : \Delta \to \Gamma$ of $\Raw$, the \emph{derived judgement forms}
    \[
        {\vdash \Gamma}, \qquad \Delta \vdash \sigma : \Gamma, \qquad {\vdash \Gamma \equiv \Gamma'}, \qquad \Delta \vdash \sigma \equiv \sigma' : \Gamma
    \]
    are defined by recursion on the length of $\Gamma$, as finite conjunctions of raw judgements, by the rules
    \begin{mathpar}
        \inferrule{ }{\vdash ()}
        \and
        \inferrule{\vdash \Gamma \\ \Gamma \vdash A}{\vdash \Gamma.A}
        \and
        \inferrule{\vdash \Delta}{\Delta \vdash \sigma : ()}
        \and
        \inferrule{\Delta \vdash \sigma_0 : \Gamma \\ \Gamma \vdash A \\ \Delta \vdash \sigma_1 : A[\sigma_0]}{\Delta \vdash \sigma : \Gamma.A}
        \\
        \inferrule{ }{\vdash () \equiv ()}
        \and
        \inferrule{\vdash \Gamma \equiv \Gamma' \\ \Gamma \vdash A \equiv A'}{\vdash \Gamma.A \equiv \Gamma'.A'}
        \and
        \inferrule{\vdash \Delta}{\Delta \vdash \sigma \equiv \sigma' : ()}
        \and
        \inferrule{\Delta \vdash \sigma_0 \equiv \sigma'_0 : \Gamma \\ \Gamma \vdash A \\ \Delta \vdash \sigma_1 \equiv \sigma'_1 : A[\sigma_0]}{\Delta \vdash \sigma \equiv \sigma' : \Gamma.A}
    \end{mathpar}
    each read as an equivalence: the conclusion holds iff its premises hold, where a premise written as a raw judgement holds if it is one and belongs to $\jdg$.
\end{definition}

Two premises are not raw judgements on their face. In the rule for $\vdash \Gamma.A \equiv \Gamma'.A'$, the type $A'$ lives over $\Gamma'$, whereas $\Gamma \vdash A \equiv A'$ requires $A' \in \Ty(\Gamma)$; in the rule for $\Delta \vdash \sigma \equiv \sigma' : \Gamma.A$, the term $\sigma'_1$ has the type $A[\sigma'_0]$, whereas $\Delta \vdash \sigma_1 \equiv \sigma'_1 : A[\sigma_0]$ requires it to have the type $A[\sigma_0]$. Both issues are resolved by~\cref{cor:sort-degeneracy}..

\begin{lemma}\label{lem:derived-forms}
    \begin{enumerate}
        \item If $\vdash \Gamma \equiv \Gamma'$, then $\Gamma^{\downarrow} = \Gamma'^{\downarrow}$ and $\Ty(\Gamma) = \Ty(\Gamma')$. Hence, given the premise $\vdash \Gamma \equiv \Gamma'$, we have $A' \in \Ty(\Gamma)$, and the premise $\Gamma \vdash A \equiv A'$ is a raw judgement precisely when $|A| = |A'|$.
        \item For $\sigma, \sigma' : \Delta \to \Gamma.A$, $\Tm(\Delta, A[\sigma'_0]) = \Tm(\Delta, A[\sigma_0])$. Hence the premise $\Delta \vdash \sigma_1 \equiv \sigma'_1 : A[\sigma_0]$ is a raw judgement.
    \end{enumerate}
    All other premises are raw judgements by the conventions on $\sigma_0$ and $\sigma_1$, so every derived form is a finite conjunction of memberships in $\jdg$.
\end{lemma}

\begin{proof}
    (1) By induction on the length of $\Gamma$. For $\vdash () \equiv ()$ there is nothing to prove. If $\vdash \Gamma.A \equiv \Gamma'.A'$, then $\Gamma^{\downarrow} = \Gamma'^{\downarrow}$ by induction, and $\Gamma \vdash A \equiv A'$ is a raw judgement, so $|A| = |A'|$; hence $(\Gamma.A)^{\downarrow} = (\Gamma'.A')^{\downarrow}$ by the definition of $(\_)^{\downarrow}$. Then $\Ty(\Gamma) = M(\Gamma^{\downarrow}) = \Ty(\Gamma')$ by \cref{cor:sort-degeneracy}, so $A' \in \Ty(\Gamma)$, and $(\Gamma, A, A') \in B_{\mathrm{eq}}$ iff $|A| = |A'|$ by \cref{def:raw-judgements}.

    (2) The shape map $|\_|$ is natural, so $A[\sigma_0]$ and $A[\sigma'_0]$ are types over $\Delta$ of the same shape $|A|$, and the two term sets agree by \cref{cor:sort-degeneracy}.
\end{proof}

\begin{lemma}\label{lem:admissible}
    Every typing is closed under the rules of \cref{fig:admissible}.
\end{lemma}

\begin{proof}
    \textsc{Conv-Tm} is \textsc{Refl-Tm}, \textsc{Conv-TmEq} and \textsc{Presup-TmEq}.

    \textsc{Refl-Ctx}, \textsc{Refl-Sb} and \textsc{Presup-SbEq}, by induction on the length of $\Gamma$. For $\Gamma = ()$ all three conclusions are $\vdash \Delta$ or $\vdash ()$. For $\Gamma.A$: the two reflexivities follow from the induction hypothesis and \textsc{Refl-Ty}, resp.\ \textsc{Refl-Tm}; and $\Delta \vdash \sigma \equiv \sigma' : \Gamma.A$ gives $\Delta \vdash \sigma_0 : \Gamma$ by induction, $\Gamma \vdash A$, and $\Delta \vdash \sigma_1 : A[\sigma_0]$ by \textsc{Presup-TmEq}.

    \textsc{Id}: for $\Gamma = ()$ this is $\vdash ()$; for $\Gamma.A$, $\id_{\Gamma.A} = \langle p_A, \nu_A \rangle$ by \cref{lem:pairing}, and $\Gamma.A \vdash p_A : \Gamma$, $\Gamma \vdash A$, $\Gamma.A \vdash \nu_A : A[p_A]$ hold by \textsc{Wk} and \textsc{Var}. \textsc{Comp}, by induction on the length of $\Gamma$: for $\Gamma.A$, $(\sigma \circ \tau)_0 = \sigma_0 \circ \tau$ and $(\sigma \circ \tau)_1 = \sigma_1[\tau]$ by \cref{lem:pairing}, so $\Xi \vdash \sigma_0 \circ \tau : \Gamma$ by induction and $\Xi \vdash \sigma_1[\tau] : A[\sigma_0][\tau] = A[\sigma_0 \circ \tau]$ by \textsc{Subst-Tm}. \textsc{Comp-Eq} is the same with \textsc{Cong-Tm} in place of \textsc{Subst-Tm}.

    \textsc{Presup-CtxEq}, its mirror image $\vdash \Gamma'$, \textsc{Id-CtxEq}, \textsc{CtxConv-Ty}, \textsc{CtxConv-Tm}, \textsc{CtxConv-TyEq} and \textsc{CtxConv-TmEq}, simultaneously by induction on the length of $\Gamma$. For $\vdash () \equiv ()$ there is nothing to prove. Let $\vdash \Gamma.A \equiv \Gamma'.A'$, that is, $\vdash \Gamma \equiv \Gamma'$ and $\Gamma \vdash A \equiv A'$, and let $\iota_0 : \Gamma' \to \Gamma$ be given by $\id_{\Gamma^{\downarrow}}$; by induction $\vdash \Gamma$, $\vdash \Gamma'$, $\Gamma' \vdash \iota_0 : \Gamma$, and judgements over $\Gamma$ transfer to $\Gamma'$. By \textsc{Presup-TyEq} and \textsc{Sym-Ty}, $\Gamma \vdash A$ and $\Gamma \vdash A'$, hence $\vdash \Gamma.A$, and by transfer $\Gamma' \vdash A'$, hence $\vdash \Gamma'.A'$, and $\Gamma' \vdash A' \equiv A$ by \textsc{Sym-Ty} and transfer. Let $\iota : \Gamma'.A' \to \Gamma.A$ be given by $\id_{(\Gamma.A)^{\downarrow}}$. By \cref{prop:raw-natural-model}, $\iota_0' \coloneqq p_A \circ \iota$ is given by the same morphism of $\Kl(T)$ as $\iota_0 \circ p_{A'}$, and $\nu_A[\iota] = \nu_{A'}$. Now $\Gamma'.A' \vdash \iota_0 \circ p_{A'} : \Gamma$ by \textsc{Wk} and \textsc{Comp}, $\Gamma \vdash A$, and $\Gamma'.A' \vdash \nu_{A'} : A[\iota_0 \circ p_{A'}] = A[p_{A'}]$: indeed $\Gamma'.A' \vdash \nu_{A'} : A'[p_{A'}]$ by \textsc{Var}, $\Gamma'.A' \vdash A'[p_{A'}] \equiv A[p_{A'}]$ by \textsc{Cong-Ty} applied to $\Gamma' \vdash A' \equiv A$ and $\Gamma'.A' \vdash p_{A'} \equiv p_{A'} : \Gamma'$ (from \textsc{Wk} and \textsc{Refl-Sb}), and \textsc{Conv-Tm} applies. So $\Gamma'.A' \vdash \iota : \Gamma.A$. Transfer: since $\iota$ is an identity of $\Kl(T)$, $B[\iota] = B$ and $u[\iota] = u$ for all types $B$ and terms $u$ over $\Gamma.A$, so \textsc{Subst-Ty} and \textsc{Subst-Tm} along $\iota$ transfer $\Gamma.A \vdash B$ and $\Gamma.A \vdash u : B$, and \textsc{Cong-Ty} and \textsc{Cong-Tm} along $\iota \equiv \iota$ (from \textsc{Refl-Sb}) transfer the equalities.

    \textsc{Sym-Ctx} and \textsc{Trans-Ctx}, by induction on the length: for $\vdash \Gamma.A \equiv \Gamma'.A'$, symmetry of the last component is \textsc{Sym-Ty} followed by \textsc{CtxConv-TyEq}; for $\vdash \Gamma.A \equiv \Gamma'.A'$ and $\vdash \Gamma'.A' \equiv \Gamma''.A''$, transfer $\Gamma' \vdash A' \equiv A''$ to $\Gamma$ by \textsc{Sym-Ctx} and \textsc{CtxConv-TyEq}, and use \textsc{Trans-Ty}. \textsc{Sym-Sb} and \textsc{Trans-Sb} are the same with \textsc{Sym-Tm} and \textsc{Trans-Tm}, and with \textsc{Cong-Ty} (from \textsc{Refl-Ty}) and \textsc{Conv-TmEq} used to compare $\sigma'_1 \equiv \sigma''_1$ at $A[\sigma'_0]$ with the required type $A[\sigma_0]$.

    \textsc{CtxConv-Sb} and \textsc{CtxConv-SbEq}, by induction on the length of $\Gamma$, using \textsc{CtxConv-Tm} and \textsc{CtxConv-TmEq} to transfer the component judgements from $\Delta$ to $\Delta'$, and \textsc{Cong-Ty} with $\Gamma \vdash A \equiv A'$ and $\Delta \vdash \sigma_0 \equiv \sigma_0 : \Gamma$ (\textsc{Refl-Sb}) followed by \textsc{Conv-Tm} and \textsc{Conv-TmEq} to pass from $A[\sigma_0]$ to $A'[\sigma_0]$.
\end{proof}

\begin{lemma}\label{lem:typing}
    Let $\jdg$ be a typing.
    \begin{enumerate}
        \item $\vdash \Gamma \equiv \Gamma'$ is an equivalence relation on the contexts $\Gamma$ with $\vdash \Gamma$, and $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ is an equivalence relation on the morphisms $\sigma$ with $\Delta \vdash \sigma : \Gamma$.
        \item If $\vdash \Delta \equiv \Delta'$ and $\vdash \Gamma \equiv \Gamma'$, then $\Delta \vdash \sigma : \Gamma$ iff $\Delta' \vdash \sigma : \Gamma'$, and $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ iff $\Delta' \vdash \sigma \equiv \sigma' : \Gamma'$, for morphisms $\sigma, \sigma' : \Gamma^{\downarrow} \to \Delta^{\downarrow}$ of $\Kl(T)$.
    \end{enumerate}
\end{lemma}

\begin{proof}
    (1) \textsc{Refl-Ctx}, \textsc{Sym-Ctx} and \textsc{Trans-Ctx}, with \textsc{Presup-CtxEq} for the domain; likewise \textsc{Refl-Sb}, \textsc{Sym-Sb}, \textsc{Trans-Sb} and \textsc{Presup-SbEq}. (2) \textsc{CtxConv-Sb} and \textsc{CtxConv-SbEq}, with \textsc{Sym-Ctx} for the converse directions.
\end{proof}

\begin{definition}\label{def:rule}
    A \emph{rule} in the raw judgements of $\Raw$ is an implication $P_1 \wedge \cdots \wedge P_n \Rightarrow Q$, $n \in \N$, whose \emph{premises} $P_i$ and \emph{conclusion} $Q$ are raw judgements; a derived form of \cref{def:derived-forms} may stand for its finite conjunction of raw judgements, as a premise or, giving finitely many rules, as a conclusion. A set of judgements is \emph{closed} under a set $X$ of rules if it contains the conclusion of every rule of $X$ whose premises it contains.
\end{definition}

\begin{proposition}\label{prop:lattice}
    For every set $X$ of rules there is a least typing $\jdg_X$ closed under $X$. It is $\bigcup_{n \in \N} \jdg_n$, where $\jdg_0 \coloneqq \varnothing$ and $\jdg_{n+1}$ is $\jdg_n$ together with the conclusions of all rules of \cref{fig:typing} and of $X$ whose premises lie in $\jdg_n$; so a raw judgement lies in $\jdg_X$ iff it has a finite derivation by these rules.
\end{proposition}

\begin{proof}
    Typings closed under $X$ are closed under arbitrary intersections, since if the premises of a rule lie in every member of a family of closed sets, so does its conclusion, and the set of all raw judgements is one. Hence their intersection is the least one. It contains $\bigcup_n \jdg_n$, by induction on $n$; conversely $\bigcup_n \jdg_n$ is closed, since a rule has finitely many premises, which then lie in some $\jdg_n$, so its conclusion lies in $\jdg_{n+1}$.
\end{proof}

\begin{definition}\label{def:well-formed}
    Let $\jdg$ be a typing on $\Raw$. A context $\Gamma$, a substitution $\sigma : \Delta \to \Gamma$, a type $A$ over $\Gamma$ and a term $t$ of $A$ are \emph{well-formed} if, respectively,
    \[
        {\vdash \Gamma}, \qquad \Delta \vdash \sigma : \Gamma, \qquad \Gamma \vdash A, \qquad \Gamma \vdash t : A.
    \]
    The well-formed contexts and substitutions form $\cat{W}_\jdg$, with
    \[
        \Ty_{\cat{W}_\jdg}(\Gamma) \coloneqq \{\, A \mid \Gamma \vdash A \,\}, \qquad \Tm_{\cat{W}_\jdg}(\Gamma, A) \coloneqq \{\, t \mid \Gamma \vdash t : A \,\}.
    \]
    The quotient $\cat{W}_\jdg/{\equiv}$ has as objects the classes $[\Gamma]$ of well-formed contexts under $\vdash \Gamma \equiv \Gamma'$, and
    \begin{gather*}
        \Hom([\Delta], [\Gamma]) \coloneqq \{\, \sigma \mid \Delta \vdash \sigma : \Gamma \,\} / (\Delta \vdash \sigma \equiv \sigma' : \Gamma), \qquad [\sigma] \circ [\tau] \coloneqq [\sigma \circ \tau],\\
        \Ty_{\cat{W}_\jdg/{\equiv}}([\Gamma]) \coloneqq \Ty_{\cat{W}_\jdg}(\Gamma) / (\Gamma \vdash A \equiv A'), \qquad [A][\sigma] \coloneqq [A[\sigma]],\\
        \Tm_{\cat{W}_\jdg/{\equiv}}([\Gamma]) \coloneqq \{\, (A, t) \mid \Gamma \vdash t : A \,\} / {\sim},\\
        p([A, t]) \coloneqq [A], \qquad [A, t][\sigma] \coloneqq [A[\sigma], t[\sigma]],
    \end{gather*}
    where $(A, t) \sim (A', t')$ iff $\Gamma \vdash A \equiv A'$ and $\Gamma \vdash t \equiv t' : A$, so that
    \[
        \Tm_{\cat{W}_\jdg/{\equiv}}([\Gamma], [A]) = \Tm_{\cat{W}_\jdg}(\Gamma, A) / (\Gamma \vdash t \equiv t' : A),
    \]
    whose elements we write $[t]$; and
    \[
        [\Gamma].[A] \coloneqq [\Gamma.A], \qquad p_{[A]} \coloneqq [p_A], \qquad \nu_{[A]} \coloneqq [\nu_A], \qquad \langle [\sigma], [t] \rangle \coloneqq [\langle \sigma, t \rangle].
    \]
\end{definition}

\begin{proposition}\label{prop:quotient}
    Let $\jdg$ be a typing on $\Raw$.
    \begin{enumerate}
        \item $\cat{W}_\jdg$ is a contextual sub-natural-model of $\Raw$.
        \item $\cat{W}_\jdg/{\equiv}$ is well defined and a contextual natural model, and $[\_] : \cat{W}_\jdg \to \cat{W}_\jdg/{\equiv}$ is a fibrewise surjective morphism of natural models.
        \item $\cat{W}_\jdg/{\equiv}$ is shaped by $|[A]| \coloneqq |A|$, and $[\_]$ is shape-preserving, so that $(\cat{W}_\jdg, [\_])$ is a subquotient of $\Raw$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    (1) The closure conditions of \cref{def:sub-natural-model}: $\vdash ()$; identities and composites by \textsc{Id} and \textsc{Comp}; closure of types and terms under the actions by \textsc{Subst-Ty} and \textsc{Subst-Tm}; $p$ restricts by \textsc{Presup-Tm}; extension by the definition of $\vdash \Gamma.A$; $p_A$ and $\nu_A$ by \textsc{Wk} and \textsc{Var}; pairing by the definition of $\Delta \vdash \langle \sigma, t \rangle : \Gamma.A$, since $\langle \sigma, t \rangle_0 = \sigma$ and $\langle \sigma, t \rangle_1 = t$. It is contextual: for $\Gamma = (A_1, \ldots, A_n)$, $\vdash \Gamma$ unfolds to $\Gamma_{<i} \vdash A_i$ for every $i$, so $\Gamma = ().A_1.\cdots.A_n$ in $\cat{W}_\jdg$, uniquely because this holds in $\Raw$.

    (2) \emph{Well-definedness.} The relations on contexts, substitutions and types are equivalence relations, by \cref{lem:typing}(1) and by \textsc{Refl-Ty}, \textsc{Sym-Ty} and \textsc{Trans-Ty}. So is $\sim$: for $\Gamma \vdash A \equiv A'$ we have $\Tm_{\cat{W}_\jdg}(\Gamma, A) = \Tm_{\cat{W}_\jdg}(\Gamma, A')$ by \textsc{Conv-Tm}, and $\Gamma \vdash t \equiv t' : A$ iff $\Gamma \vdash t \equiv t' : A'$ by \textsc{Conv-TmEq}; together with \textsc{Refl-Tm}, \textsc{Sym-Tm} and \textsc{Trans-Tm} this gives reflexivity, symmetry and transitivity of $\sim$, and the fibre of $p$ over $[A]$ is $\Tm_{\cat{W}_\jdg}(\Gamma, A)/(\Gamma \vdash t \equiv t' : A)$, as stated. The hom-sets, types and terms do not depend on the representative contexts: for $\vdash \Gamma \equiv \Gamma'$, we have $\Ty_{\cat{W}_\jdg}(\Gamma) = \Ty_{\cat{W}_\jdg}(\Gamma')$ with the same relation, and likewise for terms and for substitutions, by the \textsc{CtxConv} rules of \cref{lem:admissible}. The identifications are the identity on the raw data. Composition is well defined by \textsc{Comp-Eq}, the actions on types and terms by \textsc{Cong-Ty} and \textsc{Cong-Tm}, extension by the definition of $\vdash \Gamma.A \equiv \Gamma'.A'$, and $p_{[A]}$, $\nu_{[A]}$ because $p_A$ and $\nu_A$ depend only on $\Gamma^{\downarrow}$ and $|A|$ by \cref{prop:raw-natural-model}. Associativity, unit laws, functoriality of the actions and naturality of $p$ hold for representatives, hence for classes. The object $[()]$ is terminal.

    \emph{Representability.} We have $p(\nu_{[A]}) = [A[p_A]] = [A][p_{[A]}]$. Let $[\sigma] : [\Delta] \to [\Gamma]$ and $[t] \in \Tm_{\cat{W}_\jdg/{\equiv}}([\Delta], [A][\sigma]) = \Tm_{\cat{W}_\jdg}(\Delta, A[\sigma])/{\equiv}$; by the above we may take $t$ with $\Delta \vdash t : A[\sigma]$. Then $\Delta \vdash \langle \sigma, t \rangle : \Gamma.A$ by (1), and $\langle \sigma, t \rangle \equiv \langle \sigma', t' \rangle$ whenever $\sigma \equiv \sigma'$ and $t \equiv t'$, by \cref{def:derived-forms}; so $\langle [\sigma], [t] \rangle$ is well defined, and $p_{[A]} \circ \langle [\sigma], [t] \rangle = [\sigma]$ and $\nu_{[A]}[\langle [\sigma], [t] \rangle] = [t]$ hold for representatives. If $[\tau]$ satisfies $p_{[A]} \circ [\tau] = [\sigma]$ and $\nu_{[A]}[[\tau]] = [t]$, then $\tau = \langle \tau_0, \tau_1 \rangle$ by \cref{lem:pairing}, with $\Delta \vdash \tau_0 \equiv \sigma : \Gamma$, and $\Delta \vdash \tau_1 \equiv t : A[\sigma]$, since $\Delta \vdash A[\tau_0] \equiv A[\sigma]$ by \textsc{Cong-Ty} and \textsc{Refl-Ty}, and $\equiv$ at $A[\tau_0]$ and at $A[\sigma]$ agree by \textsc{Conv-TmEq}; hence $\Delta \vdash \tau \equiv \langle \sigma, t \rangle : \Gamma.A$.

    \emph{Contextuality.} For $\Gamma = (A_1, \ldots, A_n)$, $[\Gamma] = [()].[A_1].\cdots.[A_n]$, and $[\Gamma.A] = [\Gamma'.A']$ iff $\vdash \Gamma \equiv \Gamma'$ and $\Gamma \vdash A \equiv A'$, iff $[\Gamma] = [\Gamma']$ and $[A] = [A']$; objects of different lengths are distinct.

    \emph{The quotient map} preserves $\pt$, extension, $p_A$, $\nu_A$ and the actions by definition. It is fibrewise surjective: every element of $\Ty_{\cat{W}_\jdg/{\equiv}}([\Gamma])$ is $[A]$ for some $A \in \Ty_{\cat{W}_\jdg}(\Gamma)$, and every element of $\Tm_{\cat{W}_\jdg/{\equiv}}([\Gamma], [A])$ is $[t]$ for some $t \in \Tm_{\cat{W}_\jdg}(\Gamma, A)$, by the independence of representatives.

    (3) $\Gamma \vdash A \equiv A'$ requires $|A| = |A'|$ by \cref{def:raw-judgements}, so $|[A]| \coloneqq |A|$ is well defined; it is natural because $|A[\sigma]| = |A|$, and $[\_]$ preserves it by construction. Now (1)--(3) are the description of \cref{def:shaped-subquotient}.
\end{proof}

\begin{proposition}\label{prop:span}
    For a typing $\jdg$, in the kernel of $[\_] : \cat{W}_\jdg \twoheadrightarrow \cat{W}_\jdg/{\equiv}$ is given by the relation $\equiv$, that is, $[\Gamma] = [\Gamma']$ iff $\vdash \Gamma \equiv \Gamma'$; $[\sigma] = [\sigma']$ iff $\Delta \vdash \sigma \equiv \sigma' : \Gamma$; $[A] = [A']$ iff $\Gamma \vdash A \equiv A'$; $[t] = [t']$ iff $\Gamma \vdash t \equiv t' : A$.
\end{proposition}

\begin{proof}
    By \cref{prop:quotient} and the definitions.
\end{proof}

\begin{definition}\label{def:saturated}
    Let $(\cat{W}, q)$ be a subquotient of $\Raw$. If $q(\Gamma) = q(\Gamma')$, then $\Gamma^{\downarrow} = \Gamma'^{\downarrow}$ by \cref{lem:shaped-fiberwise-surjection}, and so $\Ty_{\Raw}(\Gamma) = \Ty_{\Raw}(\Gamma')$ by \cref{cor:sort-degeneracy}. If moreover $q_{\Ty}(\Gamma, A) = q_{\Ty}(\Gamma', A')$, then it follows by $q$'s shape preservation,
    $$|A|_{\cat{W}} = |q_{\Ty}^{\Gamma}(A)|_{\cat{Q}} = |q_{\Ty}^{\Gamma'}(A')|_{\cat{Q}} = |A'|_{\cat{W}},$$
    that $|A| = |A'|$, and thus $\Tm_{\Raw}(\Gamma, A) = \Tm_{\Raw}(\Gamma', A')$ by \cref{cor:sort-degeneracy}.
    
    The subquotient is \emph{saturated} if:
    \begin{description}[font=\normalfont, leftmargin=!, labelwidth=3em]
        \item[(S1)] if $q(\Gamma) = q(\Gamma')$, then $\Ty_\cat{W}(\Gamma) = \Ty_\cat{W}(\Gamma')$, and $q_{\Ty}(\Gamma, B) = q_{\Ty}(\Gamma', B)$ for every $B$ in them;
        \item[(S2)] if $q(\Gamma) = q(\Gamma')$ and $q_{\Ty}(\Gamma, A) = q_{\Ty}(\Gamma', A')$, then we have $\Tm_\cat{W}(\Gamma, A) = \Tm_\cat{W}(\Gamma', A')$, and $q_{\Tm}(\Gamma, A, t) = q_{\Tm}(\Gamma', A', t)$ for every $t$ in them.
    \end{description}
\end{definition}

\begin{lemma}\label{lem:saturated}
    A subquotient $(\cat{W}, q)$ of $\Raw$ is saturated iff
    \begin{enumerate}
        \item whenever $q(\Gamma) = q(\Gamma')$, the canonical isomorphism $\iota : \Gamma' \to \Gamma$ of \cref{cor:sort-degeneracy} lies in $\cat{W}$ and $q(\iota) = \id_{q\Gamma}$;
        \item whenever $q_\Ty(\Gamma, A) = q_\Ty(\Gamma, A')$, we have $\Tm_\cat{W}(\Gamma, A) = \Tm_\cat{W}(\Gamma, A')$ and $q_\Tm(\Gamma, A, t) = q_\Tm(\Gamma, A', t)$ for every $t$ in them.
    \end{enumerate}
\end{lemma}

\begin{proof}
    Suppose (1) and (2). For (S1), let $q(\Gamma) = q(\Gamma')$ and $B \in \Ty_\cat{W}(\Gamma)$; then $B = B[\iota] \in \Ty_\cat{W}(\Gamma')$ and $q_{\Ty}(\Gamma', B) = q_{\Ty}(\Gamma, B)[q\iota] = q_{\Ty}(\Gamma, B)$, and symmetrically with the inverse canonical isomorphism. For (S2), let moreover $q_\Ty(\Gamma, A) = q_\Ty(\Gamma', A')$ and $t \in \Tm_\cat{W}(\Gamma, A)$; then $t = t[\iota] \in \Tm_\cat{W}(\Gamma', A)$ with $q_\Tm(\Gamma', A, t) = q_\Tm(\Gamma, A, t)$, and since $q_\Ty(\Gamma', A) = q_\Ty(\Gamma, A) = q_\Ty(\Gamma', A')$, (2) at $\Gamma'$ gives $t \in \Tm_\cat{W}(\Gamma', A')$ with the same image; symmetrically.

    Conversely, (2) is (S2) with $\Gamma = \Gamma'$. For (1), by induction on the length, the case of the empty contexts being trivial: let $q(\Gamma.A) = q(\Gamma'.A')$ and let $\iota : \Gamma'.A' \to \Gamma.A$ and $\iota_0 : \Gamma' \to \Gamma$ be the canonical isomorphisms. By \cref{prop:trivial-fibrations}(2), $q(\Gamma) = q(\Gamma')$ and $q_\Ty(\Gamma, A) = q_\Ty(\Gamma', A')$, so $\iota_0 \in \cat{W}$ with $q\iota_0 = \id$ by induction, and $A' \in \Ty_\cat{W}(\Gamma)$ by (S1). By \cref{prop:raw-natural-model}, $p_A \circ \iota = \iota_0 \circ p_{A'}$ and $\nu_A[\iota] = \nu_{A'}$, so $\iota = \langle \iota_0 \circ p_{A'}, \nu_{A'} \rangle$ by \cref{lem:pairing}. Now $\nu_{A'} \in \Tm_\cat{W}(\Gamma'.A', A'[p_{A'}])$, and
    \begin{align*}
        q_\Ty(\Gamma'.A', A'[p_{A'}]) &= q_\Ty(\Gamma', A')[qp_{A'}] = q_\Ty(\Gamma, A)[q\iota_0 \circ qp_{A'}]\\
        &= q_\Ty(\Gamma, A)[q(\iota_0 \circ p_{A'})] = q_\Ty(\Gamma'.A', A[\iota_0 \circ p_{A'}]),
    \end{align*}
    so by (2) $\nu_{A'} \in \Tm_\cat{W}(\Gamma'.A', A[\iota_0 \circ p_{A'}])$, and $\iota \in \cat{W}$ by the closure of $\cat{W}$ under pairing. Finally $q\iota = \langle q\iota_0 \circ p_{qA'}, \nu_{qA'} \rangle = \langle p_{qA'}, \nu_{qA'} \rangle = \id$ by \cref{lem:map-of-adjunctions,lem:pairing}.
\end{proof}

\begin{remark}\label{rem:saturated}
    Saturation refers to the canonical isomorphisms and to the common raw terms of \cref{cor:sort-degeneracy}, structure of $\Raw$ that $\mathsf{CtxNM}/\cat{B}(S^+)$ does not see; it is therefore a property of subquotients of $\Raw$, not part of the notion of subquotient. It says that the identifications made by $q$ respect those made by the raw syntax itself. In the reading of \cref{prop:characterisation} below it is the conversion rule, and it is exactly what makes the kernel of $q$ be generated by equalities $\Gamma \vdash A \equiv A'$ and $\Gamma \vdash t \equiv t' : A$ over single contexts.
\end{remark}

\begin{proposition}\label{prop:characterisation}
    The assignment $\jdg \mapsto (\cat{W}_\jdg, [\_] : \cat{W}_\jdg \to \cat{W}_\jdg/{\equiv})$ is a bijection between typings on $\Raw$ and isomorphism classes of saturated subquotients of $\Raw$, with inverse $[(\cat{W}, q)] \mapsto \jdg_q$, where $\jdg_q$ consists of the $\Gamma \vdash A$ with $A \in \Ty_\cat{W}(\Gamma)$, the $\Gamma \vdash t : A$ with $t \in \Tm_\cat{W}(\Gamma, A)$, the $\Gamma \vdash A \equiv A'$ with $A, A' \in \Ty_\cat{W}(\Gamma)$ and $q_{\Ty}(\Gamma, A) = q_{\Ty}(\Gamma, A')$, and the $\Gamma \vdash t \equiv t' : A$ with $t, t' \in \Tm_\cat{W}(\Gamma, A)$ and $q_{\Tm}(\Gamma, A, t) = q_{\Tm}(\Gamma, A, t')$.
\end{proposition}

\begin{proof}
    Let $\jdg$ be a typing. By \cref{prop:quotient}, $(\cat{W}_\jdg, [\_])$ is a subquotient of $\Raw$; it is saturated, since by \cref{prop:quotient} the sets $\Ty_{\cat{W}_\jdg/{\equiv}}([\Gamma])$ and $\Tm_{\cat{W}_\jdg/{\equiv}}([\Gamma], [A])$ are computed from any representatives, with the identifications the identity on raw data, which is (S1) and (S2); and $\jdg_{[\_]} = \jdg$ by \cref{prop:span}.

    Conversely, let $(\cat{W}, q)$ be saturated. The set $\jdg_q$ depends only on the isomorphism class of $(\cat{W}, q)$: if $q' = \theta \circ q$ with $\theta$ an isomorphism, then $q'_{\Ty}(\Gamma, A) = q'_{\Ty}(\Gamma, A')$ iff $q_{\Ty}(\Gamma, A) = q_{\Ty}(\Gamma, A')$, since $\theta$ is injective on types, and likewise for terms, so $\jdg_{q'} = \jdg_q$. By contextuality of $\cat{W}$, \cref{prop:trivial-fibrations}(2) and induction on the length, the derived forms of $\jdg_q$ are: $\vdash \Gamma$ iff $\Gamma \in \cat{W}$; $\Delta \vdash \sigma : \Gamma$ iff $\sigma \in \cat{W}(\Delta, \Gamma)$; $\vdash \Gamma \equiv \Gamma'$ iff $q(\Gamma) = q(\Gamma')$; $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ iff $q(\sigma) = q(\sigma')$. We verify that $\jdg_q$ is a typing. \textsc{Presup-Ty}, \textsc{Presup-Tm}, \textsc{Presup-TyEq}, \textsc{Presup-TmEq}, \textsc{Subst-Ty}, \textsc{Subst-Tm}, \textsc{Wk} and \textsc{Var} hold because $\cat{W}$ is a sub-natural-model and the equalities of $\jdg_q$ relate elements of $\cat{W}$. \textsc{Refl-Ty}, \textsc{Sym-Ty}, \textsc{Trans-Ty}, \textsc{Refl-Tm}, \textsc{Sym-Tm} and \textsc{Trans-Tm} hold because the relations are kernels of functions. \textsc{Cong-Ty}: $q(A[\sigma]) = q(A)[q(\sigma)] = q(A')[q(\sigma')] = q(A'[\sigma'])$, and \textsc{Cong-Tm} likewise. \textsc{Conv-TmEq}: if $t \in \Tm_\cat{W}(\Gamma, A)$ and $q_{\Ty}(\Gamma, A) = q_{\Ty}(\Gamma, A')$, then $t \in \Tm_\cat{W}(\Gamma, A')$ and $q_{\Tm}(\Gamma, A', t) = q_{\Tm}(\Gamma, A, t)$ by (S2) with $\Gamma = \Gamma'$; hence $q_{\Tm}(\Gamma, A, t) = q_{\Tm}(\Gamma, A, t')$ implies $q_{\Tm}(\Gamma, A', t) = q_{\Tm}(\Gamma, A', t')$. Now $\cat{W}_{\jdg_q} = \cat{W}$, and the congruence of $\jdg_q$ is the kernel of $q$. The map $\theta : \cat{W}/{\equiv} \to \cat{Q}$, $[x] \mapsto q(x)$, is well defined and a morphism of natural models, since $q$ is and respects its kernel; it is injective on objects, faithful and fibrewise injective by \cref{prop:span}, and fibrewise surjective because $q$ is, hence also surjective on objects and full by \cref{prop:trivial-fibrations}(2). So $\theta$ is an isomorphism by \cref{lem:iso}, shape-preserving since $|\theta[A]| = |qA| = |A| = |[A]|$, with $q = \theta \circ [\_]$, and the two assignments are inverse up to isomorphism.
\end{proof}

\section{The type theory \texorpdfstring{$\ToS^\omega$}{ToS-omega}}
\label{sec:type-theories}

\begin{definition}\label{def:type-theory}
    Let $T$ be a relative monad on $J_S$. A \emph{type theory} $\TT$ over $T$ is specified by the following data:
    \begin{itemize}
        \item a module of raw types $M : \Kl(T) \to \Set^S$,
        \item a module of raw equations $E : \Kl(T) \to \Set$, and
        \item a set $X$ of rules of $\Raw(T, M, E)$.
    \end{itemize}
    The \emph{judgements} of $\TT$, $\jdg_{\TT}$ are defined as least typing \cref{prop:lattice} on $\Raw(T, M, E)$ closed under $X$.
    The \emph{syntax} of $\TT$ is the $S^+$-shaped contextual natural model $\Syn_{\TT} \coloneqq \cat{W}_{\jdg_{\TT}}/{\equiv_{\TT}}$ of \cref{prop:quotient}.
\end{definition}

In the rest of this section we define $\ToS^\omega$ as a type theory over the relative monad of expressions of \cref{ch:raw-syntax}. Its raw types are hypothetical judgements: a conclusion, which is a family of types, an element of a type, or an equation, under a telescope of hypotheses, which are again hypothetical judgements. The rules of \cref{fig:tos} say when such a judgement is well formed, when two of them are equal, and what counts as evidence for one.

\subsection{Raw syntax}
\label{sec:tos-raw}
\providecommand{\Tel}{\mathsf{Tel}}

The expressions of \cref{ch:raw-syntax} do not distinguish families of types from their elements. For $\ToS^\omega$ we record this distinction in the arities, by a kind attached to every slot.

\begin{definition}\label{def:kinds}
    Let $K \coloneqq \{\mathsf{ty}, \mathsf{tm}\}$ be the set of \emph{kinds}, and define $\widetilde{\bA}$ as the initial algebra of $X \mapsto \List(X \times K)$: an \emph{arity} is a list of \emph{shapes}, and a shape is a pair $(\alpha, \kappa)$ of an arity and a kind. The corresponding slot functor is
    \begin{align*}
        \_ &\ni\_ = J_{\widetilde{\bA}} : \mathbb{F}{\downarrow}\widetilde{\bA} \to \Set^{\widetilde{\bA}}\\
        [(\gamma_1,\lambda_1),\cdots,(\gamma_n,\lambda_n)] &\ni(\alpha,\kappa) = \{i : \{1,\cdots,n\} \mid (\gamma_i,\lambda_i) = (\alpha,\kappa)\}.
    \end{align*}
    The sets $\Expr(\Gamma, \kappa)$ of \emph{expressions of kind $\kappa$} over an arity $\Gamma$ are defined inductively by a single rule: for every slot $x : \Gamma \ni (\alpha, \kappa)$ and every family of arguments
    \[
        t \in \Args(\Gamma, \alpha) \coloneqq \prod_{(\delta, \lambda) \in S}\, \prod_{i : \alpha \ni (\delta, \lambda)} \Expr(\Gamma \concat \delta, \lambda),
    \]
    there is an expression $x(t) \in \Expr(\Gamma, \kappa)$. An expression thus has the kind of its head, and the argument filling a slot of shape $(\delta, \lambda)$ has kind $\lambda$. We have a functor
    \begin{align*}
        &\widetilde{\bE} :  \mathbb{F}{\downarrow}\widetilde{\bA} \to \Set^{\widetilde{\bA}}\\
        &\widetilde{\bE}(\Gamma,(\alpha,\kappa)) := \Expr(\Gamma \concat \alpha, \kappa).
    \end{align*}
\end{definition}

\begin{proposition}\label{prop:kinded-monad}
    With $\eta$-expansion of (\cref{eta-expansion}) as the unit, $\eta_{\Gamma,(\alpha,\kappa)} := \eta_{\Gamma,\alpha}$, and the action of \cref{def:action} as the Kleisli extension, $\sigma^\sharp_{(\alpha, \kappa)} \coloneqq \_[\sigma]^\alpha$, $\widetilde{\bE}$ is a relative monad on $J_S$.
\end{proposition}

An expression of $\Expr(\Gamma \concat \Phi, \kappa)$ may be regarded as an element of $\widetilde{\bE}(\Gamma)_{(\Phi, \kappa)}$ or of $\widetilde{\bE}(\Gamma \concat \Phi)_{(\emptyar, \kappa)}$, depending on how its arity is split, and the following lemma says that the two readings carry compatible actions. Recall from \cref{lem:kleisli-coproducts} the coproducts of $\Kl(\widetilde{\bE})$, the copairing $\langle f, g \rangle$, and $f \concat \Phi = \langle \inl \odot f, \inr \rangle$.

\begin{lemma}\label{lem:currying}
    Let $\sigma : \Gamma \to \Delta \in \Kl(\widetilde{\bE})$ be a substitution and $\Phi$ an arity. For every shape $(\alpha, \kappa)$, the square below commutes.
    \[\begin{tikzcd}
    	{\widetilde{\mathbb{E}}(\Gamma\concat\Phi)_{(\alpha,\kappa)}} && {\widetilde{\mathbb{E}}(\Delta\concat\Phi)_{(\alpha,\kappa)}} \\
    	{\widetilde{\mathbb{E}}(\Gamma)_{(\Phi\concat \alpha,\kappa)}} && {\widetilde{\mathbb{E}}(\Delta)_{(\Phi\concat \alpha,\kappa)}}
    	\arrow["{\_[\sigma\concat\Phi]^{\alpha}}", from=1-1, to=1-3]
    	\arrow[equals, from=1-1, to=2-1]
    	\arrow[equals, from=1-3, to=2-3]
    	\arrow["{\_[\sigma]^{\Phi\concat\alpha}}"', from=2-1, to=2-3]
    \end{tikzcd}\]    
\end{lemma}

\begin{proof}
    By associativity.
\end{proof}

\begin{lemma}\label{lem:instantiation}

\end{lemma}

\begin{proof}
    Easy.
\end{proof}

\begin{definition}\label{def:hypothetical}
    The families of sets $\mathfrak{C}_0, \mathfrak{C} : \bA \to \Set$ of \emph{non-equational conclusions} and of \emph{conclusions} are
    \begin{align*}
        \mathfrak{C}_0(\Gamma) &\coloneqq \{\mathsf{type}\} \cup \{\, B \mid B \in \Expr(\Gamma, \mathsf{ty}) \,\},\\
        \mathfrak{C}(\Gamma) &\coloneqq \mathfrak{C}_0(\Gamma) \cup \{\, l =_c r \mid c \in \mathfrak{C}_0(\Gamma),\ l, r \in \Expr(\Gamma, \kappa_c) \,\},
    \end{align*}
    where $\kappa : \mathfrak{C}_0(\Gamma) \to K$, the \emph{kind} of a non-equational conclusion, is given by $\kappa_{\mathsf{type}} \coloneqq \mathsf{ty}$ and $\kappa_B \coloneqq \mathsf{tm}$. Here $\mathsf{type}$, the \emph{type expressions} $B \in \Expr(\Gamma, \mathsf{ty})$ and the equations $l =_c r$ are formal symbols, so that the unions are disjoint; we use the letters $c, c'$ for non-equational conclusions and $C, C'$ for arbitrary ones. The families of sets of \emph{telescopes}, of \emph{slot entries} and of \emph{equational entries},
    \[
        \Tel : \left|\bA\right| \times \left|\bA\right| \to \Set, \qquad M : \left|\bA\right| \to \Set^S, \qquad E : \left|\bA\right| \to \Set,
    \]
    are defined by simultaneous induction, inductively by the clauses
    \begin{alignat*}{2}
        & && () \in \Tel(\Gamma, \emptyar),\\
        \Theta \in \Tel(\Gamma, \vartheta),\ A \in M(\Gamma \concat \vartheta)_s &&\implies{}& (\Theta, A) \in \Tel(\Gamma, \vartheta \concat [s]),\\
        \Theta \in \Tel(\Gamma, \vartheta),\ A \in E(\Gamma \concat \vartheta) &&\implies{}& (\Theta, A) \in \Tel(\Gamma, \vartheta),\\
        \Theta \in \Tel(\Gamma, \vartheta),\ c \in \mathfrak{C}_0(\Gamma \concat \vartheta) &&\implies{}& [\Theta]c \in M(\Gamma)_{(\vartheta, \kappa_c)},\\
        \Theta \in \Tel(\Gamma, \vartheta),\ C \in \mathfrak{C}(\Gamma \concat \vartheta) \setminus \mathfrak{C}_0(\Gamma \concat \vartheta) &&\implies{}& [\Theta]C \in E(\Gamma),
    \end{alignat*}
    for all arities $\Gamma$ and $\vartheta$, and shapes $s$.
    The elements of $M(\Gamma)_s$ and of $E(\Gamma)$ are the \emph{entries} over $\Gamma$, written $[\Theta]C$. Every telescope $\Theta$ lies in $\Tel(\Gamma, \vartheta)$ for exactly one $\vartheta$, its \emph{arity} $\Theta^{\downarrow} \coloneqq \vartheta$, and we identify a conclusion $C$ with the entry $[()]C$.
\end{definition}

An entry $[\Theta]C$ is a hypothetical judgement: the conclusion $C$ under the hypotheses $\Theta$. The conclusions internalise the four forms of judgement of the object theories~\cite{MartinLof1984}: a type is a term of $\mathsf{type}$, an element of the type $B$ is a term of the conclusion $B$, and the two forms of equality judgement are the equations $l =_{\mathsf{type}} r$ and $l =_B r$. A slot entry $[\Theta]c$ declares a slot of shape $(\Theta^{\downarrow}, \kappa_c)$ that records the slots it binds and whether it is a family or an element. An equational entry declares no slot, so the arity of a telescope counts only its slot entries. Entries generalise the abstracted boundaries by which finitary type theories declare the metavariables of a rule~\cite[\S2.1]{HaselwarterBauer2023}: their hypotheses are again entries, of arbitrary rank, and may be families and equations as well as elements.

\begin{definition}\label{def:entry-action}
    For a substitution $\sigma : \Gamma \to \Delta \in \Kl(\widetilde{\bE})$, the maps
    \begin{gather*}
        \_[\sigma] : \mathfrak{C}(\Gamma) \to \mathfrak{C}(\Delta), \qquad \_[\sigma] : \Tel(\Gamma, \gamma) \to \Tel(\Delta, \gamma),\\
        \_[\sigma] : M(\Gamma)_s \to M(\Delta)_s, \qquad \_[\sigma] : E(\Gamma) \to E(\Delta),
    \end{gather*}
    are defined by simultaneous recursion:
    \begin{gather*}
        \mathsf{type}[\sigma] \coloneqq \mathsf{type}, \qquad B[\sigma] \coloneqq \sigma^\sharp(B), \qquad (l =_c r)[\sigma] \coloneqq l[\sigma] =_{c[\sigma]} r[\sigma],\\
        ()[\sigma] \coloneqq (), \qquad (\Theta, A)[\sigma] \coloneqq (\Theta[\sigma], A[\sigma \concat \Theta^{\downarrow}]),\\
        ([\Theta]C)[\sigma] \coloneqq [\Theta[\sigma]]\, C[\sigma \concat \Theta^{\downarrow}].
    \end{gather*}
\end{definition}

\begin{lemma}\label{lem:entry-modules}
    The actions of \cref{def:entry-action} the slot entries $M$ assemble a module of raw types, and the equational entries $E$ assemble into a module of raw equations:
    \begin{align*}
        M : \Kl(\widetilde{\bE}) &\to \Set^{\widetilde{A}}\\
        E : \Kl(\widetilde{\bE}) &\to \Set.
    \end{align*}  
\end{lemma}

\begin{proof}
    Easy.
\end{proof}

Henceforth, $\Raw \coloneqq \Raw(\widetilde{\bE}, M, E)$ is the raw natural model of $\ToS^\omega$, an object of $\mathsf{CtxNM}/\cat{B}({\widetilde{\bA}}^+)$ by \cref{def:raw-natural-model}. By \cref{prop:raw-natural-model}(1), a morphism $\sigma : \Delta \to \Gamma$ of $\Raw$ is a morphism $\Gamma^{\downarrow} \to \Delta^{\downarrow}$ of $\Kl(\widetilde{\bE})$, and we denote both by $\sigma$.

\begin{definition}\label{def:telescope-operations}
    Let $\Gamma$ be a raw context and $\vartheta$ an arity.
    \begin{enumerate}
        \item The \emph{extension} of $\Gamma$ by telescopes is the map
        \[
            \Gamma.\_ : \Tel(\Gamma^{\downarrow}, \vartheta) \to \Ob \Raw, \qquad \Gamma.() \coloneqq \Gamma, \qquad \Gamma.(\Theta, A) \coloneqq (\Gamma.\Theta).A,
        \]
        defined by recursion on telescopes; here $(\Gamma.\Theta).A$ is defined because, by induction, $(\Gamma.\Theta)^{\downarrow} = \Gamma^{\downarrow} \concat \Theta^{\downarrow}$, so that $A$ is a type over $\Gamma.\Theta$ by \cref{prop:raw-natural-model}(2). The \emph{projection} of $\Gamma.\Theta$ is the morphism $p_\Theta : \Gamma.\Theta \to \Gamma$ given by $p_{()} \coloneqq \id_\Gamma$ and $p_{(\Theta, A)} \coloneqq p_\Theta \circ p_A = \inl : \Gamma^{\downarrow} \to \Gamma^{\downarrow}$.
        \item For a morphism $\sigma : \Delta \to \Gamma$ and $\Theta \in \Tel(\Gamma^{\downarrow}, \vartheta)$, let $\Theta[\sigma] \in \Tel(\Delta^{\downarrow}, \vartheta)$ be the action of \cref{def:entry-action}. The \emph{lift} of $\sigma$ along $\Theta$ is the morphism
        \[
            \sigma^{+\Theta} \coloneqq \sigma \concat \vartheta : \Delta.\Theta[\sigma] \to \Gamma.\Theta,
        \]
        that is, the morphism $\sigma \concat \vartheta : \Gamma^{\downarrow} \concat \vartheta \to \Delta^{\downarrow} \concat \vartheta$ of $\Kl(\widetilde{\bE})$ of \cref{lem:kleisli-coproducts}.
        \item For $\Theta \in \Tel(\Gamma^{\downarrow}, \vartheta)$, an \emph{instantiation} of $\Theta$ is a family of arguments $t \in \Args(\Gamma^{\downarrow}, \vartheta) = \Kl(\widetilde{\bE})(\vartheta, \Gamma^{\downarrow})$, and $\langle \id, t \rangle : \Gamma \to \Gamma.\Theta$ is the morphism $\langle \eta_{\Gamma^{\downarrow}}, t \rangle : \Gamma^{\downarrow} \concat \vartheta \to \Gamma^{\downarrow}$.
    \end{enumerate}
\end{definition}

\begin{lemma}\label{lem:telescopes}
    Let $\Gamma$ be a raw context, $\sigma : \Delta \to \Gamma$ a morphism and $\Theta \in \Tel(\Gamma^{\downarrow}, \vartheta)$.
    \begin{enumerate}
        \item $p_\Theta = \inl : \Gamma^{\downarrow} \to \Gamma^{\downarrow} \concat \vartheta$.
        \item $\sigma^{+()} = \sigma$, and $\sigma^{+(\Theta, A)} = \langle \sigma^{+\Theta} \circ p_{A'}, \nu_{A'} \rangle$, where $A' \coloneqq A[\sigma^{+\Theta}]$ is the last entry of $(\Theta, A)[\sigma] = (\Theta[\sigma], A')$.
        \item For $C \in \mathfrak{C}(\Gamma^{\downarrow} \concat \vartheta)$, $([\Theta]C)[\sigma] = [\Theta[\sigma]]\, C[\sigma^{+\Theta}]$, and the square
        \[\begin{tikzcd}
            {\Tm(\Gamma, [\Theta]C)} && {\Tm(\Delta, ([\Theta]C)[\sigma])} \\
            {\Tm(\Gamma.\Theta, C)} && {\Tm(\Delta.\Theta[\sigma], C[\sigma^{+\Theta}])}
            \arrow["{\_[\sigma]}", from=1-1, to=1-3]
            \arrow[equals, from=1-1, to=2-1]
            \arrow[equals, from=1-3, to=2-3]
            \arrow["{\_[\sigma^{+\Theta}]}"', from=2-1, to=2-3]
        \end{tikzcd}\]
        commutes.
    \end{enumerate}
\end{lemma}

\begin{proof}
    By \cref{prop:raw-natural-model}(4), $p_A = \inl$ and $\nu_A = \inr$ for a slot entry $A$ of shape $s$, and $p_A = \id$ and $\nu_A = \star$ for an equational one. (1) By induction on $\Theta$, composites of coprojections being coprojections. (2) $\sigma^{+()} = \sigma \concat \emptyar = \sigma$, the equation for $(\Theta, A)[\sigma]$ is \cref{def:entry-action}, and $\langle \sigma^{+\Theta} \circ p_{A'}, \nu_{A'} \rangle$ is $\langle \inl \odot (\sigma \concat \vartheta), \inr \rangle = \sigma \concat (\vartheta \concat [s])$ for $A$ a slot entry of shape $s$, and $\sigma \concat \vartheta$ for $A$ equational; in both cases this is $\sigma^{+(\Theta, A)}$. (3) The first equation is \cref{def:entry-action}; with it, the right side of the square is the left side at $\Delta$ and $\Theta[\sigma]$. For $C = c$ non-equational, both term sets on the left are $\Expr(\Gamma^{\downarrow} \concat \vartheta, \kappa_c)$ by \cref{prop:raw-natural-model}(3), acted on by $\_[\sigma]^{\vartheta}$ and $\_[\sigma \concat \vartheta]^{\emptyar}$, which agree by \cref{lem:currying}; for $C$ an equation, both are $\{\star\}$.
\end{proof}

\begin{proposition}\label{prop:tos-raw}
    The raw natural model $\Raw$ of $\ToS^\omega$ is as follows.
    \begin{enumerate}
        \item \emph{Objects.} The raw contexts are the telescopes over the empty arity,
        $$\Ob \Raw = \textstyle\bigsqcup_{\vartheta} \Tel(\emptyar, \vartheta).$$
        \item \emph{Morphisms.} A morphism $\sigma : \Delta \to \Gamma$ is a substitution $\sigma : \Gamma^{\downarrow} \to \Delta^{\downarrow}$, that is, a morphism of $\Kl(\widetilde{\bE})$; identities and composition are those of $\Kl(\widetilde{\bE})^{\op}$.
        \item \emph{Types.} The types over $\Gamma$ are the total space of entries over $\Gamma^{\downarrow}$,
        $$\Ty(\Gamma) = M(\Gamma^{\downarrow}) \sqcup E(\Gamma^{\downarrow}),$$
        with action defined pointwise as in \cref{def:entry-action}. The shaping is given by
        \[\begin{tikzcd}[column sep=scriptsize]
        	{\bigsqcup_sM(\Gamma)_s} & {\bigsqcup_sM(\Gamma)_s \sqcup E(\Gamma)} & {E(\Gamma)} \\
        	{\widetilde{\mathbb{A}}} & {\widetilde{\mathbb{A}}^+} & {1.}
        	\arrow[hook, from=1-1, to=1-2]
        	\arrow["\fst"', from=1-1, to=2-1]
        	\arrow["{|\_|}", from=1-2, to=2-2]
        	\arrow[hook', from=1-3, to=1-2]
        	\arrow[from=1-3, to=2-3]
        	\arrow[hook, from=2-1, to=2-2]
        	\arrow[hook', from=2-3, to=2-2]
        \end{tikzcd}\]
        \item \emph{Terms.} The terms of a slot entry are the expressions over the arity extended by its telescope, of the kind of its conclusion, and an equational entry has the single term $\star$:
        \[
            \Tm(\Gamma, [\Theta]c) = \Expr(\Gamma^{\downarrow} \concat \Theta^{\downarrow}, \kappa_c), \qquad \Tm(\Gamma, [\Theta](l =_c r)) = \{\star\}.
        \]
        A substitution $\sigma : \Delta \to \Gamma$ acts on $e : \Tm(\Gamma, [\Theta]c)$ by $e[\sigma] = e[\sigma]^{\Theta^{\downarrow}}$, and on equational terms by $\star[\sigma] = \star$.
        \item \emph{Context extension.} For a type $A$ over $\Gamma$, $\Gamma.A$ is $\Gamma$ extended by $A$. If $A \in M(\Gamma^{\downarrow})_s$, then the projection is $p_A = \inl : \Gamma^{\downarrow} \to \Gamma^{\downarrow} \concat [s]$, the generic term is $\nu_A = \inr \in \Kl(\widetilde{\bE})([s], \Gamma^{\downarrow} \concat [s]) = \widetilde{\bE}(\Gamma^{\downarrow} \concat [s])_s$, and for $\sigma : \Delta \to \Gamma$ and $u \in \Tm(\Delta, A[\sigma]) = \widetilde{\bE}(\Delta^{\downarrow})_s = \Kl(\widetilde{\bE})([s], \Delta^{\downarrow})$, the pairing $\langle \sigma, u \rangle$ is the copairing $\langle \sigma, u \rangle : \Gamma^{\downarrow} \concat [s] \to \Delta^{\downarrow}$ of \cref{lem:kleisli-coproducts}. If $A \in E(\Gamma^{\downarrow})$, then $p_A = \id_{\Gamma^{\downarrow}}$, $\nu_A = \star$, and $\langle \sigma, \star \rangle = \sigma$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    This is \cref{prop:raw-natural-model} for $T = \widetilde{\bE}$. By its part (1), the objects are the sequences $(A_1, \ldots, A_n)$ with $A_i \in M(\Gamma_{<i}^{\downarrow}) \sqcup E(\Gamma_{<i}^{\downarrow})$, and $(\_)^{\downarrow}$ appends $|A_i|$ for $A_i \in M$ and nothing for $A_i \in E$; by \cref{def:hypothetical}, these are the telescopes over $\emptyar$, with $\Gamma^{\downarrow}$ their arity. This gives (1), and the description of morphisms in the same part gives (2). Part (2) of \cref{prop:raw-natural-model}, with the shaping of \cref{def:raw-natural-model}, gives (3). Its part (3) gives $\Tm(\Gamma, [\Theta]c) = \widetilde{\bE}(\Gamma^{\downarrow})_{(\Theta^{\downarrow}, \kappa_c)} = \Expr(\Gamma^{\downarrow} \concat \Theta^{\downarrow}, \kappa_c)$, with the action $\sigma^\sharp = \_[\sigma]^{\Theta^{\downarrow}}$ of \cref{prop:kinded-monad}, which is (4). Its part (4) is (5), with the identifications of \cref{lem:kleisli-coproducts}(1).
\end{proof}

\subsection{Rules}
\label{sec:tos-rules}

A raw type of $\ToS^\omega$ is a hypothetical judgement $[\Theta]C$, and by \cref{lem:telescopes}(3) evidence for it, a term of $[\Theta]C$ over $\Gamma$, is literally evidence for its conclusion $C$ over $\Gamma.\Theta$. The rules of $\ToS^\omega$, collected in \cref{fig:tos}, accordingly fall into two groups: the rules for hypotheses, which are uniform in the conclusion, and the rules for the three conclusions, at the empty telescope.

\emph{Hypotheses.} A hypothetical judgement is well formed if its conclusion is well formed under its hypotheses (\textsc{Bind-Ty}), and two of them are equal if their hypotheses and their conclusions are (\textsc{Bind-TyEq}). Evidence for a hypothetical judgement is evidence for its conclusion under its hypotheses, and so is equality of evidence (\textsc{Bind-Tm}, \textsc{Bind-TmEq}); these two rules relate judgements about the same raw terms. Binding is thus not a syntactic operation: abstraction and application are identities, and $\beta$ and $\eta$ hold on the nose, which is how the expressions of $\ToS^\omega$ stay $\beta$-normal and $\eta$-long (\cref{rem:normal-forms}).

\emph{Conclusions.} The conclusion $\mathsf{type}$ is well formed in every well-formed context (\textsc{Type-Ty}). The conclusion $B$, of being an element of $B$, is well formed if $B$ is a type (\textsc{Elem-Ty}), and two such conclusions are equal if the two types are (\textsc{Elem-TyEq}). An equation is well formed if its two sides are evidence for its conclusion $c$ (\textsc{Eqn-Ty}), and two equations are equal if their conclusions and sides are (\textsc{Eqn-TyEq}). Finally, $\star$ is evidence for an equation precisely when its two sides are judgementally equal (\textsc{Eqn-Tm}): read downwards, this is reflexivity, and read upwards, it reflects an equation into a judgemental equality. There are no rules for evidence of $\mathsf{type}$ or of $B$: $\ToS^\omega$ has no constructors of types or of elements, all of which are declared as slots of a context and typed through them (\textsc{App} of \cref{fig:tos-admissible}). In terms of natural models, $\mathsf{type}$ is a universe, the conclusion $B$ its decoding, $l =_c r$ an extensional identity type and $[\Theta]\_$ a dependent product over $\Theta$ with strict $\beta$ and $\eta$.

Note that Only \textsc{Bind-Tm} and \textsc{Eqn-Tm} are stated in both directions. The converses of the other rules hold by \cref{prop:inversion} below, because a type is never a hypothesis: a context is extended by types whose well-formedness has been derived, so every derivation of a judgement about an entry $[\Theta]C$, or about two entries, rests on derivations of the corresponding judgements about $\Theta$ and $C$. Evidence, on the other hand, can be hypothetical. The rule \textsc{Var} of \cref{fig:typing} makes the slot $x$ declared by a slot entry $[\Theta]c$ into evidence $\nu = \eta(x)$ for $[\Theta]c$, and an equational entry into evidence $\star$ for itself, without any derivation over the extended context. The upward directions of \textsc{Bind-Tm} and \textsc{Eqn-Tm} are therefore not inversions but the elimination rules of hypotheses: the first is how a slot is applied (\textsc{App}), the second how an equation is used (\textsc{Hyp}). The converse of \textsc{Bind-TmEq} is again admissible: \textsc{Var} provides evidence but no equalities, and the equalities that equational hypotheses yield, through the upward direction of \textsc{Eqn-Tm}, lie at the empty telescope, where the converse is trivial. Lastly, no premise is redundant: the well-formedness of contexts and conclusions is presupposed (\textsc{Presup-Ty}, \textsc{Presup-Tm}), so that, for instance, \textsc{Bind-Ty} has no premise $\vdash \Gamma.\Theta$ and \textsc{Eqn-Ty} no premise $\Gamma \vdash c$. The one premise of this kind is that of \textsc{Type-Ty}, without which \textsc{Presup-Ty} would make every raw context well formed. An equation records the conclusion $c$ of its sides because the judgement $\Gamma \vdash l \equiv r : c$ in \textsc{Eqn-Tm} does.

\begin{definition}\label{def:tos}
    The \emph{theory of signatures} $\ToS^\omega$ is the type theory over $\bE$ with the module of raw types $M$ and the module of raw equations $E$ of \cref{def:hypothetical}, and with the rules of \cref{fig:tos} (a rule drawn with a double line is two rules, read downwards and upwards). Its judgements $\jdg_{\ToS^\omega}$ are thus the raw judgements of $\Raw$ derivable by the rules of \cref{fig:typing,fig:tos} (\cref{prop:lattice}); in the rest of this section, a raw judgement written on its own asserts membership in $\jdg_{\ToS^\omega}$.
\end{definition}

\subsection{Admissible rules}
\label{sec:tos-admissible}

We first record how the extensions $\sigma^{+\Theta}$ of \cref{lem:telescopes}(2) behave in a typing.

\begin{lemma}\label{lem:lifting}
    Let $\jdg$ be a typing on $\Raw$, let $\sigma, \sigma' : \Delta \to \Gamma$, and let $\Theta, \Theta'$ be telescopes over $\Gamma^{\downarrow}$.
    \begin{enumerate}
        \item If $\Delta \vdash \sigma : \Gamma$ and $\vdash \Gamma.\Theta$, then $\vdash \Delta.\Theta[\sigma]$ and $\Delta.\Theta[\sigma] \vdash \sigma^{+\Theta} : \Gamma.\Theta$.
        \item If $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ and $\Gamma \vdash \Theta \equiv \Theta'$, then $\Delta \vdash \Theta[\sigma] \equiv \Theta'[\sigma']$ and $\Delta.\Theta[\sigma] \vdash \sigma^{+\Theta} \equiv \sigma'^{+\Theta'} : \Gamma.\Theta$.
    \end{enumerate}
    In (2), $\Theta^{\downarrow} = \Theta'^{\downarrow}$ by \cref{lem:derived-forms}, so $\sigma'^{+\Theta'}$ is also a morphism $\Delta.\Theta[\sigma] \to \Gamma.\Theta$ by \cref{cor:sort-degeneracy}.
\end{lemma}

\begin{proof}
    By induction on the length of $\Theta$; for $\Theta = ()$ the claims are the hypotheses, together with $\vdash \Delta \equiv \Delta$, which follows from $\vdash \Delta$ by \textsc{Refl-Ctx}. Let $\Theta = (\Theta_0, A)$ and $\tau \coloneqq \sigma^{+\Theta_0}$. (1) By $\vdash \Gamma.\Theta_0$ and induction, $\vdash \Delta.\Theta_0[\sigma]$ and $\Delta.\Theta_0[\sigma] \vdash \tau : \Gamma.\Theta_0$, so $\Gamma.\Theta_0 \vdash A$ and \textsc{Subst-Ty} give $\Delta.\Theta_0[\sigma] \vdash A[\tau]$, that is, $\vdash \Delta.\Theta[\sigma]$ by \cref{lem:telescopes}(2). The components of $\sigma^{+\Theta} = \langle \tau \circ p_{A[\tau]}, \nu_{A[\tau]} \rangle$ are well formed by \textsc{Wk}, \textsc{Comp} and \textsc{Var}, so $\Delta.\Theta[\sigma] \vdash \sigma^{+\Theta} : \Gamma.\Theta$ by \cref{def:derived-forms}. (2) Let $\Theta' = (\Theta'_0, A')$ and $\tau' \coloneqq \sigma'^{+\Theta'_0}$. By induction, $\Delta \vdash \Theta_0[\sigma] \equiv \Theta'_0[\sigma']$ and $\Delta.\Theta_0[\sigma] \vdash \tau \equiv \tau' : \Gamma.\Theta_0$, and \textsc{Cong-Ty} applied to $\Gamma.\Theta_0 \vdash A \equiv A'$ gives $\Delta.\Theta_0[\sigma] \vdash A[\tau] \equiv A'[\tau']$, so $\Delta \vdash \Theta[\sigma] \equiv \Theta'[\sigma']$. Since $|A[\tau]| = |A'[\tau']|$, the projections $p_{A[\tau]}$ and $p_{A'[\tau']}$ are given by the same morphism of $\Kl(\bE)$, and so are $\nu_{A[\tau]}$ and $\nu_{A'[\tau']}$ (\cref{prop:raw-natural-model}(4)); hence \textsc{Comp-Eq} with \textsc{Refl-Sb}, and \textsc{Refl-Tm}, give the components of $\Delta.\Theta[\sigma] \vdash \sigma^{+\Theta} \equiv \sigma'^{+\Theta'} : \Gamma.\Theta$.
\end{proof}

\begin{proposition}[Inversion]\label{prop:inversion}
    The converses of \textup{\textsc{Bind-Ty}}, \textup{\textsc{Bind-TyEq}}, \textup{\textsc{Bind-TmEq}}, \textup{\textsc{Elem-Ty}}, \textup{\textsc{Elem-TyEq}}, \textup{\textsc{Eqn-Ty}} and \textup{\textsc{Eqn-TyEq}} hold in $\ToS^\omega$:
    \begin{enumerate}
        \item if $\Gamma \vdash [\Theta]C$, then $\Gamma.\Theta \vdash C$; if $\Gamma \vdash B$, then $\Gamma \vdash B : \mathsf{type}$; and if $\Gamma \vdash l =_c r$, then $\Gamma \vdash l : c$ and $\Gamma \vdash r : c$;
        \item if $\Gamma \vdash [\Theta]C \equiv [\Theta']C'$, then $\Gamma \vdash \Theta \equiv \Theta'$ and $\Gamma.\Theta \vdash C \equiv C'$; if $\Gamma \vdash B \equiv B'$, then $\Gamma \vdash B \equiv B' : \mathsf{type}$; and if $\Gamma \vdash (l =_c r) \equiv (l' =_{c'} r')$, then $\Gamma \vdash c \equiv c'$, $\Gamma \vdash l \equiv l' : c$ and $\Gamma \vdash r \equiv r' : c$;
        \item if $\Gamma \vdash u \equiv u' : [\Theta]C$, then $\Gamma.\Theta \vdash u \equiv u' : C$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    Write $\jdg \coloneqq \jdg_{\ToS^\omega}$. A conclusion $C$ is \emph{invertible} over a raw context $\Delta$ if $C = \mathsf{type}$; or $C = B$ and $\Delta \vdash B : \mathsf{type}$; or $C = (l =_c r)$, $\Delta \vdash l : c$, $\Delta \vdash r : c$, and $c$ is invertible over $\Delta$. By recursion on entries, an entry $[\Theta]C$ over $\Gamma^{\downarrow}$ is \emph{invertible} over $\Gamma$ if every entry $\Theta_i$ of $\Theta$ is invertible over $\Gamma.\Theta_{<i}$, $\Gamma.\Theta \vdash C$, and $C$ is invertible over $\Gamma.\Theta$; a raw context is invertible if each of its entries is invertible over the preceding ones, so that $\Gamma.[\Theta]C$ is invertible iff $\Gamma.\Theta.C$ is. Two entries $[\Theta]C$ and $[\Theta']C'$ over $\Gamma^{\downarrow}$ are \emph{invertibly equal} if $\Gamma \vdash \Theta \equiv \Theta'$ and $\Gamma.\Theta \vdash C \equiv C'$, and moreover $\Gamma.\Theta \vdash B \equiv B' : \mathsf{type}$ if $C = B$ and $C' = B'$, and $\Gamma.\Theta \vdash c \equiv c'$, $\Gamma.\Theta \vdash l \equiv l' : c$ and $\Gamma.\Theta \vdash r \equiv r' : c$ if $C = (l =_c r)$ and $C' = (l' =_{c'} r')$. Let $\jdg^* \subseteq \jdg$ consist of the judgements $\Gamma \vdash A$ and $\Gamma \vdash u : A$ of $\jdg$ with $\Gamma.A$ invertible; the $\Gamma \vdash A \equiv A'$ of $\jdg$ with $\Gamma.A$ and $\Gamma.A'$ invertible and $A, A'$ invertibly equal; and the $\Gamma \vdash u \equiv u' : [\Theta]C$ of $\jdg$ with $\Gamma.[\Theta]C$ invertible and $\Gamma.\Theta \vdash u \equiv u' : C$. If $\jdg^*$ is a typing closed under \cref{fig:tos}, then $\jdg^* = \jdg$ by minimality, and (1)--(3) are part of the definition of $\jdg^*$. We use two facts about $\jdg$.

    (i) \emph{Stability.} Let $\Delta \vdash \sigma : \Gamma$. If $A = [\Theta]C$ is invertible over $\Gamma$, then $A[\sigma]$ is invertible over $\Delta$. Indeed, $\vdash \Gamma.\Theta$ by \textsc{Presup-Ty}, so the extensions $\sigma^{+\Theta_{<i}}$ and $\sigma^{+\Theta}$ are well formed by \cref{lem:lifting}(1); by induction on entries, the $i$-th entry $\Theta_i[\sigma^{+\Theta_{<i}}]$ of $\Theta[\sigma]$ is invertible over $\Delta.\Theta[\sigma]_{<i}$, and \textsc{Subst-Ty} and \textsc{Subst-Tm} along $\sigma^{+\Theta}$ give $\Delta.\Theta[\sigma] \vdash C[\sigma^{+\Theta}]$ and the invertibility of $C[\sigma^{+\Theta}]$, so that $A[\sigma] = [\Theta[\sigma]]\,C[\sigma^{+\Theta}]$ is invertible. Likewise, if $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ and $A, A'$ are invertibly equal over $\Gamma$, then $A[\sigma], A'[\sigma']$ are invertibly equal over $\Delta$, by \cref{lem:lifting}(2) and \textsc{Cong-Ty} and \textsc{Cong-Tm} along $\sigma^{+\Theta} \equiv \sigma'^{+\Theta'}$; and $\Gamma.\Theta \vdash u \equiv u' : C$ gives $\Delta.\Theta[\sigma] \vdash u[\sigma] \equiv u'[\sigma'] : C[\sigma^{+\Theta}]$, by \textsc{Cong-Tm} along $\sigma^{+\Theta} \equiv \sigma'^{+\Theta}$ and \cref{lem:telescopes}(3).

    (ii) \emph{Context conversion.} If $\vdash \Gamma \equiv \Gamma'$, an entry invertible over $\Gamma$ is invertible over $\Gamma'$, by (i) along the canonical isomorphism $\iota : \Gamma' \to \Gamma$, which is well formed by \textsc{Id-CtxEq} and satisfies $A[\iota] = A$.

    We now check the rules of \cref{fig:typing,fig:tos}: whenever their premises lie in $\jdg^*$, their conclusion, which lies in $\jdg$, satisfies the conditions of $\jdg^*$. A premise that is a derived form lies in $\jdg^*$ conjunct by conjunct; in particular, $\Delta \vdash \sigma : \Gamma$ and $\Delta \vdash \sigma \equiv \sigma' : \Gamma$ contain $\vdash \Delta$, so $\Delta$ is then invertible.

    \emph{Presuppositions.} The conditions on the conclusions of \textsc{Presup-Ty}, \textsc{Presup-Tm}, \textsc{Presup-TyEq} and \textsc{Presup-TmEq} are among those on their premises, prefixes of invertible contexts being invertible.

    \emph{Substitution.} In \textsc{Subst-Ty}, \textsc{Subst-Tm}, \textsc{Cong-Ty} and \textsc{Cong-Tm}, the context $\Delta$ is invertible, and (i) applies along $\sigma$ and $\sigma'$, which are well formed by \textsc{Presup-SbEq} and \textsc{Sym-Sb}. In \textsc{Wk} and \textsc{Var}, the context $\Gamma.A$ is invertible, and (i) applies along the projections $\Gamma.A \to \Gamma_{<i}$, which are well formed by \textsc{Wk} and \textsc{Comp}, to the entries of $\Gamma$ and to $A$.

    \emph{Equivalence and conversion.} Let $A = [\Theta]C$, $A' = [\Theta']C'$ and $A'' = [\Theta'']C''$. For \textsc{Refl-Ty}, $A$ and $A$ are invertibly equal by \textsc{Refl-Ctx}, \textsc{Refl-Ty} and \textsc{Refl-Tm}. For \textsc{Sym-Ty} and \textsc{Trans-Ty}, we get $\Gamma \vdash \Theta' \equiv \Theta$ and $\Gamma \vdash \Theta \equiv \Theta''$ by \textsc{Sym-Ctx} and \textsc{Trans-Ctx}, and the remaining conditions by \textsc{Sym-Ty}, \textsc{Trans-Ty}, \textsc{Sym-Tm} and \textsc{Trans-Tm} over $\Gamma.\Theta$, after transferring judgements between $\Gamma.\Theta$, $\Gamma.\Theta'$ and $\Gamma.\Theta''$ by the \textsc{CtxConv} rules of \cref{fig:admissible}, and, for equations, judgements at $c'$ to judgements at $c$ by \textsc{Conv-TmEq}. For \textsc{Refl-Tm}, the upward direction of \textsc{Bind-Tm} turns $\Gamma \vdash u : [\Theta]C$ into $\Gamma.\Theta \vdash u : C$, to which \textsc{Refl-Tm} applies; \textsc{Sym-Tm} and \textsc{Trans-Tm} are applied over $\Gamma.\Theta$. For \textsc{Conv-TmEq}, the context $\Gamma.A'$ is invertible by the condition on $\Gamma \vdash A \equiv A'$, and $\Gamma.\Theta \vdash u \equiv u' : C$ and $\Gamma.\Theta \vdash C \equiv C'$ give $\Gamma.\Theta' \vdash u \equiv u' : C'$ by \textsc{Conv-TmEq} and \textsc{CtxConv-TmEq} along $\Gamma \vdash \Theta \equiv \Theta'$.

    \emph{Hypotheses.} For \textsc{Bind-Ty} and both directions of \textsc{Bind-Tm}, the conditions on the premise and on the conclusion both say that $\Gamma.\Theta.C$ is invertible. For \textsc{Bind-TmEq}, the conditions on the conclusion are the invertibility of $\Gamma.\Theta.C$, which is part of the condition on the premise, and the premise itself. For \textsc{Bind-TyEq}, the conditions on the conclusion follow from those on the premises, except the invertibility of $\Gamma.[\Theta']C'$. For this, the conjunct $\Gamma.\Theta_{<i} \vdash \Theta_i \equiv \Theta'_i$ of $\Gamma \vdash \Theta \equiv \Theta'$ makes $\Theta'_i$ invertible over $\Gamma.\Theta_{<i}$, hence over $\Gamma.\Theta'_{<i}$ by (ii); and $\Gamma.\Theta \vdash C'$, by \textsc{Sym-Ty} and \textsc{Presup-TyEq}, and the invertibility of $C'$ over $\Gamma.\Theta$, part of the condition on $\Gamma.\Theta \vdash C \equiv C'$, transfer to $\Gamma.\Theta'$ by \textsc{CtxConv-Ty} and (ii).

    \emph{Conclusions.} For \textsc{Type-Ty}, \textsc{Elem-Ty} and \textsc{Eqn-Ty}, the conditions are the premises, together with, for \textsc{Eqn-Ty}, the invertibility of $c$, which holds since $\Gamma.c$, the context of the premise $\Gamma \vdash l : c$, is invertible. For \textsc{Elem-TyEq} and \textsc{Eqn-TyEq}, the equality conditions are the premises, and it remains to see that $\Gamma.B$ and $\Gamma.B'$, resp.\ $\Gamma.(l =_c r)$ and $\Gamma.(l' =_{c'} r')$, are invertible. Here $\Gamma \vdash B : \mathsf{type}$ and $\Gamma \vdash B' : \mathsf{type}$, resp.\ $\Gamma \vdash l : c$, $\Gamma \vdash r : c$, $\Gamma \vdash l' : c'$ and $\Gamma \vdash r' : c'$, follow from the premises by \textsc{Sym-Tm}, \textsc{Presup-TmEq} and \textsc{Conv-Tm}; the invertibility of $c$ and $c'$ is part of the condition on $\Gamma \vdash c \equiv c'$; and the entries are well formed by \textsc{Elem-Ty}, resp.\ \textsc{Eqn-Ty}. For the downward direction of \textsc{Eqn-Tm}, $\Gamma \vdash l : c$ and $\Gamma \vdash r : c$ follow by \textsc{Presup-TmEq} and \textsc{Sym-Tm}, hence $\Gamma \vdash l =_c r$ by \textsc{Eqn-Ty}, and $c$ is invertible by the condition on the premise. For its upward direction, the condition is the invertibility of $\Gamma.c$, which follows from that of $l =_c r$ over $\Gamma$, with $\Gamma \vdash c$ by \textsc{Presup-Tm}.
\end{proof}

The familiar rules of a logical framework are admissible. For a raw context $\Gamma = (A_1, \ldots, A_n)$, the entry $A_i$ \emph{weakened to} $\Gamma$ is $A_i[\pi]$ for the projection $\pi : \Gamma \to \Gamma_{<i}$. If $A_i$ is a slot entry, we write $x : A_i[\pi] \in \Gamma$; if it is an equational entry, we write $A_i[\pi] \in \Gamma$.

    \centering
    \begin{mathpar}
        \inferrule*[right=\textsc{App}]{x : [\Theta]c \in \Gamma \\ \Gamma \vdash t : \Theta}{\Gamma \vdash x(t) : c[t]}
        \and
        \inferrule*[right=\textsc{App-Eq}]{x : [\Theta]c \in \Gamma \\ \Gamma \vdash t \equiv t' : \Theta}{\Gamma \vdash x(t) \equiv x(t') : c[t]}
        \\
        \inferrule*[right=\textsc{Hyp}]{[\Psi](l =_c r) \in \Gamma \\ \Gamma \vdash t : \Psi}{\Gamma \vdash l[t] \equiv r[t] : c[t]}
        \and
        \inferrule*[right=\textsc{Conv-Elem}]{\Gamma \vdash u : B \\ \Gamma \vdash B \equiv B' : \mathsf{type}}{\Gamma \vdash u : B'}
        \\
        \inferrule*[right=\textsc{Inst}]{\Gamma \vdash u : [\Theta]C \\ \Gamma \vdash t : \Theta}{\Gamma \vdash u[t] : C[t]}
        \and
        \inferrule*[right=\textsc{Inst-Eq}]{\Gamma \vdash u \equiv u' : [\Theta]C \\ \Gamma \vdash t \equiv t' : \Theta}{\Gamma \vdash u[t] \equiv u'[t'] : C[t]}

\begin{proposition}\label{prop:tos-admissible}
    The judgements of $\ToS^\omega$ are closed under the rules of \cref{fig:tos-admissible}. Moreover, if $\vdash \Gamma$, then $\Gamma \vdash \eta(x) : A$ for every $x : A \in \Gamma$, and $\Gamma \vdash \star : A$ for every equational $A \in \Gamma$.
\end{proposition}

\begin{proof}
    For the second claim, let $A_i$ be the entry of $\Gamma$ in question and $\pi' : \Gamma \to \Gamma_{<i}.A_i$ the projection, which is well formed by \textsc{Wk} and \textsc{Comp}. \textsc{Var} gives $\Gamma_{<i}.A_i \vdash \nu_{A_i} : A_i[p_{A_i}]$, and \textsc{Subst-Tm} along $\pi'$ gives $\Gamma \vdash \nu_{A_i}[\pi'] : A_i[\pi]$, as $p_{A_i} \circ \pi' = \pi$. If $A_i$ is equational, then $\nu_{A_i}[\pi'] = \star$. Otherwise $\nu_{A_i} = \eta(y)$ for the last slot $y$ of $(\Gamma_{<i}.A_i)^{\downarrow}$, and $\pi'$ is given by $\eta \circ J_S(\iota)$ for the inclusion $\iota$ of $(\Gamma_{<i}.A_i)^{\downarrow}$ into $\Gamma^{\downarrow}$ (\cref{lem:telescopes}(1)), so $\nu_{A_i}[\pi'] = \eta(\iota y) = \eta(x)$ by the unit law $f^\sharp \circ \eta = f$.

    In \textsc{App}, \textsc{App-Eq} and \textsc{Hyp}, $\vdash \Gamma$ is part of the premise $\Gamma \vdash t : \Theta$, resp.\ $\Gamma \vdash t \equiv t' : \Theta$ or $\Gamma \vdash t : \Psi$. \textsc{App}: by the second claim and the upward direction of \textsc{Bind-Tm}, $\Gamma.\Theta \vdash \eta(x) : c$, and \textsc{Subst-Tm} along $\langle \id, t \rangle$ gives $\Gamma \vdash \eta(x)[t] : c[t]$, where $\eta(x)[t] = \eta(x)[\langle \eta, t \rangle] = x(t)$ by \cref{lem:instantiation}. \textsc{App-Eq}: likewise, with \textsc{Refl-Tm} and \textsc{Cong-Tm} along $\langle \id, t \rangle \equiv \langle \id, t' \rangle$ in place of \textsc{Subst-Tm}. \textsc{Hyp}: by the second claim and the upward directions of \textsc{Bind-Tm} and \textsc{Eqn-Tm}, $\Gamma.\Psi \vdash l \equiv r : c$, and \textsc{Cong-Tm} along $\langle \id, t \rangle$, with \textsc{Refl-Sb}, gives the conclusion. \textsc{Conv-Elem}: \textsc{Elem-TyEq} and \textsc{Conv-Tm}. \textsc{Inst} and \textsc{Inst-Eq}: the upward direction of \textsc{Bind-Tm}, resp.\ \cref{prop:inversion}(3), followed by \textsc{Subst-Tm}, resp.\ \textsc{Cong-Tm}, along $\langle \id, t \rangle$.
\end{proof}

\begin{remark}\label{rem:app}
    Let $x : [\Theta]c \in \Gamma$ with $\Theta = (A_1, \ldots, A_n)$, and let $t \in \Args(\Gamma^{\downarrow}, \Theta^{\downarrow})$. For each $i$, write $t_{<i} \in \Args(\Gamma^{\downarrow}, \Theta_{<i}^{\downarrow})$ for the restriction of $t$ to the slots declared by $\Theta_{<i} \coloneqq (A_1, \ldots, A_{i-1})$, and, if $A_i$ is a slot entry, $t_i \in \widetilde{\bE}(\Gamma^{\downarrow})_{|A_i|}$ for the value of $t$ at the slot that $A_i$ declares, the $i$-th argument of $x$. The component of $\langle \id, t \rangle : \Gamma \to \Gamma.\Theta$ at $A_i$ is $t_i$ for a slot entry, by the unit law, and $\star$ for an equational one. So by \cref{def:derived-forms}, the premise $\Gamma \vdash t : \Theta$ of \textsc{App} holds iff $\vdash \Gamma.\Theta$ and, for every $i$,
    \begin{itemize}
        \item $\Gamma \vdash t_i : A_i[t_{<i}]$ if $A_i$ is a slot entry: the argument is evidence for the declaration of its slot, with the preceding arguments substituted; for $A_i = [\Phi_i]d_i$, this says by \textsc{Bind-Tm} that $t_i$ is evidence for $d_i$ under the hypotheses $\Phi_i$, both instantiated by $t_{<i}$;
        \item $\Gamma \vdash \star : A_i[t_{<i}]$ if $A_i$ is an equational entry: by \textsc{Bind-Tm} and \textsc{Eqn-Tm}, the equational premise $A_i$ holds for the preceding arguments.
    \end{itemize}
    In particular, a slot can only be applied to arguments that satisfy its equational premises. The conclusion $\Gamma \vdash x(t) : c[t]$ then says that the application is evidence for the declared conclusion of $x$, with all arguments substituted. \textsc{App} is the special case of \textsc{Inst} for the evidence $\eta(x)$ that the slot provides, and \textsc{Inst} is the $\beta$-rule of $[\Theta]\_$: instantiating evidence $u$ for $[\Theta]C$ by $t$ is substituting $t$ into $u$.
\end{remark}
